   396→  organizeLayout: () => {
   397→    const { nodes, edges } = get();
   398→    if (nodes.length === 0) return;
   399→    // Simple collision avoidance + radial layout
   400→    // 1. Find roots (nodes with no incoming edges)
   401→    const incoming = new Set(edges.map((e) => e.targetId));
   402→    const roots = nodes.filter((n) => !incoming.has(n.id));
   403→    const rootId = roots.length > 0 ? roots[0].id : nodes[0].id;
   404→    // 2. Build parent-child map
   405→    const childrenOf = new Map<string, string[]>();
   406→    for (const e of edges) {
   407→      if (!childrenOf.has(e.sourceId)) childrenOf.set(e.sourceId, []);
   408→      childrenOf.get(e.sourceId)!.push(e.targetId);
   409→    }
   410→    // 3. BFS to assign depth levels
   411→    const depthMap = new Map<string, number>();
   412→    const queue: Array<{ id: string; depth: number }> = [{ id: rootId, depth: 0 }];
   413→    const visited = new Set<string>();
   414→    while (queue.length) {
   415→      const { id, depth } = queue.shift()!;
   416→      if (visited.has(id)) continue;
   417→      visited.add(id);
   418→      depthMap.set(id, depth);
   419→      for (const child of childrenOf.get(id) ?? []) {
   420→        if (!visited.has(child)) queue.push({ id: child, depth: depth + 1 });
   421→      }
   422→    }
   423→    // Assign depth 0 to unvisited nodes
   424→    for (const n of nodes) {
   425→      if (!depthMap.has(n.id)) depthMap.set(n.id, 0);
   426→    }
   427→    // 4. Compute positions by depth level with radial spread
   428→    const maxDepth = Math.max(...Array.from(depthMap.values()));
   429→    const ringRadii = [0, 260, 460, 620, 760, 900, 1040];
   430→    const centerX = 400;
   431→    const centerY = 300;
   432→    // Group nodes by depth
   433→    const byDepth = new Map<number, string[]>();
   434→    for (const [id, d] of depthMap) {
   435→      if (!byDepth.has(d)) byDepth.set(d, []);
   436→      byDepth.get(d)!.push(id);
   437→    }
   438→    const newPositions = new Map<string, { x: number; y: number }>();
   439→    for (const [d, ids] of byDepth) {
   440→      const radius = d === 0 ? 0 : (ringRadii[d] ?? ringRadii[ringRadii.length - 1] + (d - ringRadii.length + 1) * 140);
   441→      const angleStep = (2 * Math.PI) / Math.max(ids.length, 1);
   442→      const startAngle = d === 0 ? 0 : -Math.PI / 2;
   443→      for (let i = 0; i < ids.length; i++) {
   444→        const angle = startAngle + i * angleStep;
   445→        const node = nodes.find((n) => n.id === ids[i]);
   446→        const w = node?.width ?? 220;
   447→        const h = node?.height ?? 88;
   448→        newPositions.set(ids[i], {
   449→          x: centerX + Math.cos(angle) * radius - w / 2,
   450→          y: centerY + Math.sin(angle) * radius - h / 2,
   451→        });
   452→      }
   453→    }
   454→    // 5. Simple collision resolution: push overlapping nodes apart
   455→    const iterations = 3;
   456→    for (let iter = 0; iter < iterations; iter++) {
   457→      for (let i = 0; i < nodes.length; i++) {
   458→        const posA = newPositions.get(nodes[i].id) ?? { x: nodes[i].x, y: nodes[i].y };
   459→        for (let j = i + 1; j < nodes.length; j++) {
   460→          const posB = newPositions.get(nodes[j].id) ?? { x: nodes[j].x, y: nodes[j].y };
   461→          const wA = nodes[i].width + 20; // padding
   462→          const hA = nodes[i].height + 20;
   463→          const wB = nodes[j].width + 20;
   464→          const hB = nodes[j].height + 20;
   465→          const overlapX = (wA / 2 + wB / 2) - Math.abs(posA.x + nodes[i].width / 2 - posB.x + nodes[j].width / 2);
   466→          const overlapY = (hA / 2 + hB / 2) - Math.abs(posA.y + nodes[i].height / 2 - posB.y + nodes[j].height / 2);
   467→          if (overlapX > 0 && overlapY > 0) {
   468→            // Push apart along the axis with less overlap
   469→            const push = overlapX < overlapY ? overlapX / 2 : overlapY / 2;
   470→            const dirX = posA.x + nodes[i].width / 2 < posB.x + nodes[j].width / 2 ? -1 : 1;
   471→            const dirY = posA.y + nodes[i].height / 2 < posB.y + nodes[j].height / 2 ? -1 : 1;
   472→            if (overlapX < overlapY) {
   473→              newPositions.set(nodes[i].id, { ...posA, x: posA.x + dirX * push });
   474→              newPositions.set(nodes[j].id, { ...posB, x: posB.x - dirX * push });
   475→            } else {
   476→              newPositions.set(nodes[i].id, { ...posA, y: posA.y + dirY * push });
   477→              newPositions.set(nodes[j].id, { ...posB, y: posB.y - dirY * push });
   478→            }
   479→          }
   480→        }
   481→      }
   482→    }
   483→    // 6. Apply new positions
   484→    const updated = nodes.map((n) => {
   485→      const pos = newPositions.get(n.id);
   486→      return pos ? { ...n, x: pos.x, y: pos.y, updatedAt: new Date().toISOString() } : n;
   487→    });
   488→    set({ nodes: updated, dirty: true });
   489→  },

   491→  applyLayout: (layoutType) => {
   492→    const { nodes, edges } = get();
   493→    if (nodes.length === 0) return;

   495→    get().pushHistory();
   496→    const results = computeLayout(layoutType, nodes, edges);
   497→    if (results.length === 0) return;
   498→    const posById = new Map(results.map((r) => [r.id, r]));
   499→    const updated = nodes.map((n) => {

   501→      return np
   502→        ? { ...n, x: np.x, y: np.y, updatedAt: new Date().toISOString() }
   503→        : n;
   504→    });
   505→    set({ nodes: updated, dirty: true });
   506→    // Note: viewport intentionally left unchanged — user can press F / use
   507→    // "Ajustar à tela" to fit the new layout to the screen.
   508→  },

   510→  insertSubtree: (template, position, parentId) => {
   511→    // Push current state to history so the insertion is undoable as a single step
   512→    get().pushHistory();
   513→
   514→    // Tree layout: root at `position`. Children are indented horizontally by
   515→    // 200px and stacked vertically with 60px spacing. Each child's subtree is
   516→    // placed below the previous sibling's subtree (so siblings don't overlap).
   517→    const NODE_HEIGHT = 80;
   518→    const NODE_WIDTH = 200;
   519→    const VERTICAL_GAP = 60;
   520→    const HORIZONTAL_INDENT = 200;
   521→
   522→    const addNode = get().addNode;
   523→    const addEdge = get().addEdge;
   524→
   525→    // Recursive helper — returns the y-coordinate of the BOTTOM of the inserted
   526→    // subtree (so the next sibling can be placed below without overlap).
   527→    const buildSubtree = (
   528→      node: SubtreeTemplateNode,
   529→      x: number,
   530→      y: number,
   531→      parent: string | null
   532→    ): number => {
   533→      const id = addNode({
   534→        title: node.title,
   535→        kind: node.kind,
   536→        content: node.content,
   537→        note: node.note,
   538→        icon: node.icon,
   539→        x,
   540→        y,
   541→        width: NODE_WIDTH,
   542→        height: NODE_HEIGHT,
   543→        parentId: parent,
   544→      });
   545→      if (parent) {
   546→        addEdge(parent, id);
   547→      }
   548→
   549→      let bottomY = y + NODE_HEIGHT;
   550→      const kids = node.children;
   551→      if (kids && kids.length > 0) {
   552→        let childY = y + NODE_HEIGHT + VERTICAL_GAP;
   553→        for (const child of kids) {
   554→          const childBottom = buildSubtree(
   555→            child,
   556→            x + HORIZONTAL_INDENT,
   557→            childY,
   558→            id
   559→          );
   560→          childY = childBottom + VERTICAL_GAP;
   561→          if (childBottom > bottomY) bottomY = childBottom;
   562→        }
   563→      }
   564→      return bottomY;
   565→    };
   566→
   567→    const rootId = addNode({
   568→      title: template.title,
   569→      kind: template.kind,
   570→      content: template.content,
   571→      note: template.note,
   572→      icon: template.icon,
   573→      x: position.x,
   574→      y: position.y,
   575→      width: NODE_WIDTH,
   576→      height: NODE_HEIGHT,
   577→      parentId: parentId ?? null,
   578→    });
   579→    if (parentId) {
   580→      addEdge(parentId, rootId);
   581→    }
   582→
   583→    // Lay out the root's children beneath it
   584→    const kids = template.children;
   585→    if (kids && kids.length > 0) {
   586→      let childY = position.y + NODE_HEIGHT + VERTICAL_GAP;
   587→      for (const child of kids) {
   588→        const childBottom = buildSubtree(
   589→          child,
   590→          position.x + HORIZONTAL_INDENT,
   591→          childY,
   592→          rootId
   593→        );
   594→        childY = childBottom + VERTICAL_GAP;
   595→      }
   596→    }
   597→
   598→    // Select the new root so the user can immediately interact with it
   599→    set({ selectedNodeIds: [rootId], selectedEdgeIds: [], dirty: true });
   600→    return rootId;
   601→  },
   602→
   603→  pushHistory: () =>
   604→    set((s) => {
   605→      const entry: HistoryEntry = { nodes: s.nodes, edges: s.edges };
   606→      const past = [...s.past, entry].slice(-50);
   607→      return { past, future: [] };
   608→    }),
   609→
   610→  undo: () =>
   611→    set((s) => {
   612→      if (s.past.length === 0) return {};
   613→      const previous = s.past[s.past.length - 1];
   614→      const current: HistoryEntry = { nodes: s.nodes, edges: s.edges };
   615→      return {
   616→        past: s.past.slice(0, -1),
   617→        future: [current, ...s.future].slice(0, 50),
   618→        nodes: previous.nodes,
   619→        edges: previous.edges,
   620→        dirty: true,
   621→      };
   622→    }),
   623→
   624→  redo: () =>
   625→    set((s) => {
   626→      if (s.future.length === 0) return {};
   627→      const next = s.future[0];
   628→      const current: HistoryEntry = { nodes: s.nodes, edges: s.edges };
   629→      return {
   630→        past: [...s.past, current].slice(-50),
   631→        future: s.future.slice(1),
   632→        nodes: next.nodes,
   633→        edges: next.edges,
   634→        dirty: true,
   635→      };
   636→    }),
   637→
   638→  // ── Reparent (Task 16-B) ──────────────────────────────────────────────
   639→  setReparentTarget: (id) => set({ reparentTargetId: id }),
   640→  setDraggedNode: (id) => set({ draggedNodeId: id }),
   641→
   642→  isDescendantOf: (nodeId, potentialAncestorId) => {
   643→    // Check if `nodeId` is a descendant of `potentialAncestorId` by traversing
   644→    // edges downward. Returns true if the node is in the subtree of the ancestor,
   645→    // which would create a cycle if we reparented the ancestor under the node.
   646→    const { edges } = get();
   647→    // Build children map (source → targets)
   648→    const childrenOf = new Map<string, string[]>();
   649→    for (const e of edges) {
   650→      if (!childrenOf.has(e.sourceId)) childrenOf.set(e.sourceId, []);
   651→      childrenOf.get(e.sourceId)!.push(e.targetId);
   652→    }
   653→    // BFS from potentialAncestorId
   654→    const visited = new Set<string>();
   655→    const queue = [potentialAncestorId];
   656→    while (queue.length) {
   657→      const cur = queue.shift()!;
   658→      if (cur === nodeId) return true; // nodeId is a descendant
   659→      if (visited.has(cur)) continue;
   660→      visited.add(cur);
   661→      for (const child of childrenOf.get(cur) ?? []) {
   662→        if (!visited.has(child)) queue.push(child);
   663→      }
   664→    }
   665→    return false;
   666→  },
   667→
   668→  reparentNode: (nodeId, newParentId) => {
   669→    // Cannot reparent to self
   670→    if (nodeId === newParentId) return false;
   671→    // Cannot reparent to own descendant (would create cycle)
   672→    if (get().isDescendantOf(newParentId, nodeId)) return false;
   673→
   674→    // Note: pushHistory() is NOT called here because the caller
   675→    // (handleNodePointerDown in MindMapCanvas) already pushes history
   676→    // before the drag starts, which captures the initial state.
   677→    // The reparent + position changes should be undoable as a single step.
   678→
   679→    // Find the old parent edge (where targetId === nodeId)
   680→    const oldEdge = get().edges.find((e) => e.targetId === nodeId);
   681→    const oldEdgeKind = oldEdge?.kind ?? "related";
   682→
   683→    // Delete old parent edge if exists
   684→    if (oldEdge) {
   685→      get().deleteEdge(oldEdge.id);
   686→    }
   687→
   688→    // Create new edge from newParentId to nodeId
   689→    get().addEdge(newParentId, nodeId, oldEdgeKind as EdgeKind);
   690→
   691→    // Optionally reposition node near its new parent (+200px right, +60px below)
   692→    const newParent = get().nodes.find((n) => n.id === newParentId);
   693→    if (newParent) {
   694→      const draggedNode = get().nodes.find((n) => n.id === nodeId);
   695→      if (draggedNode) {
   696→        // Check if there are already children of the new parent at that position
   697→        // and offset further to avoid overlap
   698→        const childrenOfNewParent = get().edges
   699→          .filter((e) => e.sourceId === newParentId && e.targetId !== nodeId)
   700→          .map((e) => get().nodes.find((n) => n.id === e.targetId));
   701→        // Calculate y offset — stack below existing children
   702→        let yOffset = 60;
   703→        for (const child of childrenOfNewParent) {
   704→          if (child) {
   705→            yOffset = Math.max(yOffset, child.y - newParent.y + child.height + 60);
   706→          }
   707→        }
   708→        get().updateNode(nodeId, {
   709→          x: newParent.x + newParent.width + 200,
   710→          y: newParent.y + yOffset,
   711→        });
   712→      }
   713→    }
   714→
   715→    // Clear reparent state
   716→    set({ reparentTargetId: null, draggedNodeId: null });
   717→    return true;
   718→  },
   719→
   720→  // ── Search & Replace (Task 15-B) ─────────────────────────────────────
   721→  setSearchQuery: (q) => set({ searchQuery: q }),
   722→  setSearchMatches: (ids) => set({ searchMatches: ids }),
   723→  setHighlightedMatch: (id) => set({ highlightedMatchId: id }),
   724→
   725→  searchNodes: (query, opts) => {
   726→    const caseSensitive = opts?.caseSensitive ?? false;
   727→    const titleOnly = opts?.titleOnly ?? false;
   728→    const q = caseSensitive ? query : query.toLowerCase();
   729→    if (!query.trim()) {
   730→      set({ searchQuery: query, searchMatches: [], highlightedMatchId: null });
   731→      return [];
   732→    }
   733→    const nodes = get().nodes;
   734→    const matched = nodes
   735→      .filter((n) => {
   736→        const title = caseSensitive ? n.title : n.title.toLowerCase();
   737→        if (title.includes(q)) return true;
   738→        if (titleOnly) return false;
   739→        const content = caseSensitive ? (n.content ?? "") : (n.content ?? "").toLowerCase();
   740→        const note = caseSensitive ? (n.note ?? "") : (n.note ?? "").toLowerCase();
   741→        return content.includes(q) || note.includes(q);
   742→      })
   743→      .map((n) => n.id);
   744→    set({
   745→      searchQuery: query,
   746→      searchMatches: matched,
   747→      highlightedMatchId: matched.length > 0 ? matched[0] : null,
   748→    });
   749→    return matched;
   750→  },
   751→
   752→  replaceInNode: (nodeId, search, replacement, opts) => {
   753→    const caseSensitive = opts?.caseSensitive ?? false;
   754→    if (!search) return 0;
   755→    const node = get().nodes.find((n) => n.id === nodeId);
   756→    if (!node) return 0;
   757→    // Build a case-sensitive or case-insensitive regex; escape regex special chars.
   758→    const flags = caseSensitive ? "g" : "gi";
   759→    const escaped = search.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   760→    const re = new RegExp(escaped, flags);
   761→    let count = 0;
   762→    const patch: Partial<MapNode> = {};
   763→    if (node.title) {
   764→      const matches = node.title.match(re);
   765→      if (matches && matches.length > 0) {
   766→        patch.title = node.title.replace(re, replacement);
   767→        count += matches.length;
   768→      }
   769→    }
   770→    if (node.content) {
   771→      const matches = node.content.match(re);
   772→      if (matches && matches.length > 0) {
   773→        patch.content = node.content.replace(re, replacement);
   774→        count += matches.length;
   775→      }
   776→    }
   777→    if (count > 0) {
   778→      get().pushHistory();
   779→      get().updateNode(nodeId, patch);
   780→    }
   781→    return count;
   782→  },
   783→
   784→  replaceAll: (search, replacement, opts) => {
   785→    const caseSensitive = opts?.caseSensitive ?? false;
   786→    if (!search) return 0;
   787→    const matchedIds = get().searchMatches;
   788→    if (matchedIds.length === 0) return 0;
   789→    const flags = caseSensitive ? "g" : "gi";
   790→    const escaped = search.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   791→    const re = new RegExp(escaped, flags);
   792→    let total = 0;
   793→    const nodes = get().nodes;
   794→    const updates = new Map<string, Partial<MapNode>>();
   795→    for (const id of matchedIds) {
   796→      const n = nodes.find((x) => x.id === id);
   797→      if (!n) continue;
   798→      const patch: Partial<MapNode> = {};
   799→      if (n.title) {
   800→        const m = n.title.match(re);
   801→        if (m && m.length > 0) {
   802→          patch.title = n.title.replace(re, replacement);
   803→          total += m.length;
   804→        }
   805→      }
   806→      if (n.content) {
   807→        const m = n.content.match(re);
   808→        if (m && m.length > 0) {
   809→          patch.content = n.content.replace(re, replacement);
   810→          total += m.length;
   811→        }
   812→      }
   813→      if (Object.keys(patch).length > 0) {
   814→        updates.set(id, patch);
   815→      }
   816→    }
   817→    if (updates.size > 0) {
   818→      get().pushHistory();
   819→      set((s) => ({
   820→        nodes: s.nodes.map((n) =>
   821→          updates.has(n.id)
   822→            ? { ...n, ...updates.get(n.id)!, updatedAt: new Date().toISOString() }
   823→            : n
   824→        ),
   825→        dirty: true,
   826→      }));
   827→    }
   828→    return total;
   829→  },
   830→}));

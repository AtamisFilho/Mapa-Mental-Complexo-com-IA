   396→   396→  organizeLayout: () => {
   397→   397→    const { nodes, edges } = get();
   398→   398→    if (nodes.length === 0) return;
   399→   399→    // Simple collision avoidance + radial layout
   400→   400→    // 1. Find roots (nodes with no incoming edges)
   401→   401→    const incoming = new Set(edges.map((e) => e.targetId));
   402→   402→    const roots = nodes.filter((n) => !incoming.has(n.id));
   403→   403→    const rootId = roots.length > 0 ? roots[0].id : nodes[0].id;
   404→   404→    // 2. Build parent-child map
   405→   405→    const childrenOf = new Map<string, string[]>();
   406→   406→    for (const e of edges) {
   407→   407→      if (!childrenOf.has(e.sourceId)) childrenOf.set(e.sourceId, []);
   408→   408→      childrenOf.get(e.sourceId)!.push(e.targetId);
   409→   409→    }
   410→   410→    // 3. BFS to assign depth levels
   411→   411→    const depthMap = new Map<string, number>();
   412→   412→    const queue: Array<{ id: string; depth: number }> = [{ id: rootId, depth: 0 }];
   413→   413→    const visited = new Set<string>();
   414→   414→    while (queue.length) {
   415→   415→      const { id, depth } = queue.shift()!;
   416→   416→      if (visited.has(id)) continue;
   417→   417→      visited.add(id);
   418→   418→      depthMap.set(id, depth);
   419→   419→      for (const child of childrenOf.get(id) ?? []) {
   420→   420→        if (!visited.has(child)) queue.push({ id: child, depth: depth + 1 });
   421→   421→      }
   422→   422→    }
   423→   423→    // Assign depth 0 to unvisited nodes
   424→   424→    for (const n of nodes) {
   425→   425→      if (!depthMap.has(n.id)) depthMap.set(n.id, 0);
   426→   426→    }
   427→   427→    // 4. Compute positions by depth level with radial spread
   428→   428→    const maxDepth = Math.max(...Array.from(depthMap.values()));
   429→   429→    const ringRadii = [0, 260, 460, 620, 760, 900, 1040];
   430→   430→    const centerX = 400;
   431→   431→    const centerY = 300;
   432→   432→    // Group nodes by depth
   433→   433→    const byDepth = new Map<number, string[]>();
   434→   434→    for (const [id, d] of depthMap) {
   435→   435→      if (!byDepth.has(d)) byDepth.set(d, []);
   436→   436→      byDepth.get(d)!.push(id);
   437→   437→    }
   438→   438→    const newPositions = new Map<string, { x: number; y: number }>();
   439→   439→    for (const [d, ids] of byDepth) {
   440→   440→      const radius = d === 0 ? 0 : (ringRadii[d] ?? ringRadii[ringRadii.length - 1] + (d - ringRadii.length + 1) * 140);
   441→   441→      const angleStep = (2 * Math.PI) / Math.max(ids.length, 1);
   442→   442→      const startAngle = d === 0 ? 0 : -Math.PI / 2;
   443→   443→      for (let i = 0; i < ids.length; i++) {
   444→   444→        const angle = startAngle + i * angleStep;
   445→   445→        const node = nodes.find((n) => n.id === ids[i]);
   446→   446→        const w = node?.width ?? 220;
   447→   447→        const h = node?.height ?? 88;
   448→   448→        newPositions.set(ids[i], {
   449→   449→          x: centerX + Math.cos(angle) * radius - w / 2,
   450→   450→          y: centerY + Math.sin(angle) * radius - h / 2,
   451→   451→        });
   452→   452→      }
   453→   453→    }
   454→   454→    // 5. Simple collision resolution: push overlapping nodes apart
   455→   455→    const iterations = 3;
   456→   456→    for (let iter = 0; iter < iterations; iter++) {
   457→   457→      for (let i = 0; i < nodes.length; i++) {
   458→   458→        const posA = newPositions.get(nodes[i].id) ?? { x: nodes[i].x, y: nodes[i].y };
   459→   459→        for (let j = i + 1; j < nodes.length; j++) {
   460→   460→          const posB = newPositions.get(nodes[j].id) ?? { x: nodes[j].x, y: nodes[j].y };
   461→   461→          const wA = nodes[i].width + 20; // padding
   462→   462→          const hA = nodes[i].height + 20;
   463→   463→          const wB = nodes[j].width + 20;
   464→   464→          const hB = nodes[j].height + 20;
   465→   465→          const overlapX = (wA / 2 + wB / 2) - Math.abs(posA.x + nodes[i].width / 2 - posB.x + nodes[j].width / 2);
   466→   466→          const overlapY = (hA / 2 + hB / 2) - Math.abs(posA.y + nodes[i].height / 2 - posB.y + nodes[j].height / 2);
   467→   467→          if (overlapX > 0 && overlapY > 0) {
   468→   468→            // Push apart along the axis with less overlap
   469→   469→            const push = overlapX < overlapY ? overlapX / 2 : overlapY / 2;
   470→   470→            const dirX = posA.x + nodes[i].width / 2 < posB.x + nodes[j].width / 2 ? -1 : 1;
   471→   471→            const dirY = posA.y + nodes[i].height / 2 < posB.y + nodes[j].height / 2 ? -1 : 1;
   472→   472→            if (overlapX < overlapY) {
   473→   473→              newPositions.set(nodes[i].id, { ...posA, x: posA.x + dirX * push });
   474→   474→              newPositions.set(nodes[j].id, { ...posB, x: posB.x - dirX * push });
   475→   475→            } else {
   476→   476→              newPositions.set(nodes[i].id, { ...posA, y: posA.y + dirY * push });
   477→   477→              newPositions.set(nodes[j].id, { ...posB, y: posB.y - dirY * push });
   478→   478→            }
   479→   479→          }
   480→   480→        }
   481→   481→      }
   482→   482→    }
   483→   483→    // 6. Apply new positions
   484→   484→    const updated = nodes.map((n) => {
   485→   485→      const pos = newPositions.get(n.id);
   486→   486→      return pos ? { ...n, x: pos.x, y: pos.y, updatedAt: new Date().toISOString() } : n;
   487→   487→    });
   488→   488→    set({ nodes: updated, dirty: true });
   489→   489→  },
   490→   490→
   491→   491→  applyLayout: (layoutType) => {
   492→   492→    const { nodes, edges } = get();
   493→   493→    if (nodes.length === 0) return;
   494→   494→    // Snapshot current state so the layout change is a single undoable step.
   495→   495→    get().pushHistory();
   496→   496→    const results = computeLayout(layoutType, nodes, edges);
   497→   497→    if (results.length === 0) return;
   498→   498→    const posById = new Map(results.map((r) => [r.id, r]));
   499→   499→    const updated = nodes.map((n) => {
   500→   500→      const np = posById.get(n.id);
   501→   501→      return np
   502→   502→        ? { ...n, x: np.x, y: np.y, updatedAt: new Date().toISOString() }
   503→   503→        : n;
   504→   504→    });
   505→   505→    set({ nodes: updated, dirty: true });
   506→   506→    // Note: viewport intentionally left unchanged — user can press F / use
   507→   507→    // "Ajustar à tela" to fit the new layout to the screen.
   508→   508→  },

   510→   510→  insertSubtree: (template, position, parentId) => {
   511→   511→    // Push current state to history so the insertion is undoable as a single step
   512→   512→    get().pushHistory();
   513→   513→
   514→   514→    // Tree layout: root at `position`. Children are indented horizontally by
   515→   515→    // 200px and stacked vertically with 60px spacing. Each child's subtree is
   516→   516→    // placed below the previous sibling's subtree (so siblings don't overlap).
   517→   517→    const NODE_HEIGHT = 80;
   518→   518→    const NODE_WIDTH = 200;
   519→   519→    const VERTICAL_GAP = 60;
   520→   520→    const HORIZONTAL_INDENT = 200;
   521→   521→
   522→   522→    const addNode = get().addNode;
   523→   523→    const addEdge = get().addEdge;
   524→   524→
   525→   525→    // Recursive helper — returns the y-coordinate of the BOTTOM of the inserted
   526→   526→    // subtree (so the next sibling can be placed below without overlap).
   527→   527→    const buildSubtree = (
   528→   528→      node: SubtreeTemplateNode,
   529→   529→      x: number,
   530→   530→      y: number,
   531→   531→      parent: string | null
   532→   532→    ): number => {
   533→   533→      const id = addNode({
   534→   534→        title: node.title,
   535→   535→        kind: node.kind,
   536→   536→        content: node.content,
   537→   537→        note: node.note,
   538→   538→        icon: node.icon,
   539→   539→        x,
   540→   540→        y,
   541→   541→        width: NODE_WIDTH,
   542→   542→        height: NODE_HEIGHT,
   543→   543→        parentId: parent,
   544→   544→      });
   545→   545→      if (parent) {
   546→   546→        addEdge(parent, id);
   547→   547→      }
   548→   548→
   549→   549→      let bottomY = y + NODE_HEIGHT;
   550→   550→      const kids = node.children;
   551→   551→      if (kids && kids.length > 0) {
   552→   552→        let childY = y + NODE_HEIGHT + VERTICAL_GAP;
   553→   553→        for (const child of kids) {
   554→   554→          const childBottom = buildSubtree(
   555→   555→            child,
   556→   556→            x + HORIZONTAL_INDENT,
   557→   557→            childY,
   558→   558→            id
   559→   559→          );
   560→   560→          childY = childBottom + VERTICAL_GAP;
   561→   561→          if (childBottom > bottomY) bottomY = childBottom;
   562→   562→        }
   563→   563→      }
   564→   564→      return bottomY;
   565→   565→    };
   566→   566→
   567→   567→    const rootId = addNode({
   568→   568→      title: template.title,
   569→   569→      kind: template.kind,
   570→   570→      content: template.content,
   571→   571→      note: template.note,
   572→   572→      icon: template.icon,
   573→   573→      x: position.x,
   574→   574→      y: position.y,
   575→   575→      width: NODE_WIDTH,
   576→   576→      height: NODE_HEIGHT,
   577→   577→      parentId: parentId ?? null,
   578→   578→    });
   579→   579→    if (parentId) {
   580→   580→      addEdge(parentId, rootId);
   581→   581→    }
   582→   582→
   583→   583→    // Lay out the root's children beneath it
   584→   584→    const kids = template.children;
   585→   585→    if (kids && kids.length > 0) {
   586→   586→      let childY = position.y + NODE_HEIGHT + VERTICAL_GAP;
   587→   587→      for (const child of kids) {
   588→   588→        const childBottom = buildSubtree(
   589→   589→          child,
   590→   590→          position.x + HORIZONTAL_INDENT,
   591→   591→          childY,
   592→   592→          rootId
   593→   593→        );
   594→   594→        childY = childBottom + VERTICAL_GAP;
   595→   595→      }
   596→   596→    }
   597→   597→
   598→   598→    // Select the new root so the user can immediately interact with it
   599→   599→    set({ selectedNodeIds: [rootId], selectedEdgeIds: [], dirty: true });
   600→   600→    return rootId;
   601→   601→  },
   602→   602→
   603→   603→  pushHistory: () =>
   604→   604→    set((s) => {
   605→   605→      const entry: HistoryEntry = { nodes: s.nodes, edges: s.edges };
   606→   606→      const past = [...s.past, entry].slice(-50);
   607→   607→      return { past, future: [] };
   608→   608→    }),
   609→   609→
   610→   610→  undo: () =>
   611→   611→    set((s) => {
   612→   612→      if (s.past.length === 0) return {};
   613→   613→      const previous = s.past[s.past.length - 1];
   614→   614→      const current: HistoryEntry = { nodes: s.nodes, edges: s.edges };
   615→   615→      return {
   616→   616→        past: s.past.slice(0, -1),
   617→   617→        future: [current, ...s.future].slice(0, 50),
   618→   618→        nodes: previous.nodes,
   619→   619→        edges: previous.edges,
   620→   620→        dirty: true,
   621→   621→      };
   622→   622→    }),
   623→   623→
   624→   624→  redo: () =>
   625→   625→    set((s) => {
   626→   626→      if (s.future.length === 0) return {};
   627→   627→      const next = s.future[0];
   628→   628→      const current: HistoryEntry = { nodes: s.nodes, edges: s.edges };
   629→   629→      return {
   630→   630→        past: [...s.past, current].slice(-50),
   631→   631→        future: s.future.slice(1),
   632→   632→        nodes: next.nodes,
   633→   633→        edges: next.edges,
   634→   634→        dirty: true,
   635→   635→      };
   636→   636→    }),
   637→   637→
   638→   638→  // ── Reparent (Task 16-B) ──────────────────────────────────────────────
   639→   639→  setReparentTarget: (id) => set({ reparentTargetId: id }),
   640→   640→  setDraggedNode: (id) => set({ draggedNodeId: id }),
   641→   641→
   642→   642→  isDescendantOf: (nodeId, potentialAncestorId) => {
   643→   643→    // Check if `nodeId` is a descendant of `potentialAncestorId` by traversing
   644→   644→    // edges downward. Returns true if the node is in the subtree of the ancestor,
   645→   645→    // which would create a cycle if we reparented the ancestor under the node.
   646→   646→    const { edges } = get();
   647→   647→    // Build children map (source → targets)
   648→   648→    const childrenOf = new Map<string, string[]>();
   649→   649→    for (const e of edges) {
   650→   650→      if (!childrenOf.has(e.sourceId)) childrenOf.set(e.sourceId, []);
   651→   651→      childrenOf.get(e.sourceId)!.push(e.targetId);
   652→   652→    }
   653→   653→    // BFS from potentialAncestorId
   654→   654→    const visited = new Set<string>();
   655→   655→    const queue = [potentialAncestorId];
   656→   656→    while (queue.length) {
   657→   657→      const cur = queue.shift()!;
   658→   658→      if (cur === nodeId) return true; // nodeId is a descendant
   659→   659→      if (visited.has(cur)) continue;
   660→   660→      visited.add(cur);
   661→   661→      for (const child of childrenOf.get(cur) ?? []) {
   662→   662→        if (!visited.has(child)) queue.push(child);
   663→   663→      }
   664→   664→    }
   665→   665→    return false;
   666→   666→  },
   667→   667→
   668→   668→  reparentNode: (nodeId, newParentId) => {
   669→   669→    // Cannot reparent to self
   670→   670→    if (nodeId === newParentId) return false;
   671→   671→    // Cannot reparent to own descendant (would create cycle)
   672→   672→    if (get().isDescendantOf(newParentId, nodeId)) return false;
   673→   673→
   674→   674→    // Note: pushHistory() is NOT called here because the caller
   675→   675→    // (handleNodePointerDown in MindMapCanvas) already pushes history
   676→   676→    // before the drag starts, which captures the initial state.
   677→   677→    // The reparent + position changes should be undoable as a single step.
   678→   678→
   679→   679→    // Find the old parent edge (where targetId === nodeId)
   680→   680→    const oldEdge = get().edges.find((e) => e.targetId === nodeId);
   681→   681→    const oldEdgeKind = oldEdge?.kind ?? "related";
   682→   682→
   683→   683→    // Delete old parent edge if exists
   684→   684→    if (oldEdge) {
   685→   685→      get().deleteEdge(oldEdge.id);
   686→   686→    }
   687→   687→
   688→   688→    // Create new edge from newParentId to nodeId
   689→   689→    get().addEdge(newParentId, nodeId, oldEdgeKind as EdgeKind);
   690→   690→
   691→   691→    // Optionally reposition node near its new parent (+200px right, +60px below)
   692→   692→    const newParent = get().nodes.find((n) => n.id === newParentId);
   693→   693→    if (newParent) {
   694→   694→      const draggedNode = get().nodes.find((n) => n.id === nodeId);
   695→   695→      if (draggedNode) {
   696→   696→        // Check if there are already children of the new parent at that position
   697→   697→        // and offset further to avoid overlap
   698→   698→        const childrenOfNewParent = get().edges
   699→   699→          .filter((e) => e.sourceId === newParentId && e.targetId !== nodeId)
   700→   700→          .map((e) => get().nodes.find((n) => n.id === e.targetId));
   701→   701→        // Calculate y offset — stack below existing children
   702→   702→        let yOffset = 60;
   703→   703→        for (const child of childrenOfNewParent) {
   704→   704→          if (child) {
   705→   705→            yOffset = Math.max(yOffset, child.y - newParent.y + child.height + 60);
   706→   706→          }
   707→   707→        }
   708→   708→        get().updateNode(nodeId, {
   709→   709→          x: newParent.x + newParent.width + 200,
   710→   710→          y: newParent.y + yOffset,
   711→   711→        });
   712→   712→      }
   713→   713→    }
   714→   714→
   715→   715→    // Clear reparent state
   716→   716→    set({ reparentTargetId: null, draggedNodeId: null });
   717→   717→    return true;
   718→   718→  },
   719→   719→
   720→   720→  // ── Search & Replace (Task 15-B) ─────────────────────────────────────
   721→   721→  setSearchQuery: (q) => set({ searchQuery: q }),
   722→   722→  setSearchMatches: (ids) => set({ searchMatches: ids }),
   723→   723→  setHighlightedMatch: (id) => set({ highlightedMatchId: id }),
   724→   724→
   725→   725→  searchNodes: (query, opts) => {
   726→   726→    const caseSensitive = opts?.caseSensitive ?? false;
   727→   727→    const titleOnly = opts?.titleOnly ?? false;
   728→   728→    const q = caseSensitive ? query : query.toLowerCase();
   729→   729→    if (!query.trim()) {
   730→   730→      set({ searchQuery: query, searchMatches: [], highlightedMatchId: null });
   731→   731→      return [];
   732→   732→    }
   733→   733→    const nodes = get().nodes;
   734→   734→    const matched = nodes
   735→   735→      .filter((n) => {
   736→   736→        const title = caseSensitive ? n.title : n.title.toLowerCase();
   737→   737→        if (title.includes(q)) return true;
   738→   738→        if (titleOnly) return false;
   739→   739→        const content = caseSensitive ? (n.content ?? "") : (n.content ?? "").toLowerCase();
   740→   740→        const note = caseSensitive ? (n.note ?? "") : (n.note ?? "").toLowerCase();
   741→   741→        return content.includes(q) || note.includes(q);
   742→   742→      })
   743→   743→      .map((n) => n.id);
   744→   744→    set({
   745→   745→      searchQuery: query,
   746→   746→      searchMatches: matched,
   747→   747→      highlightedMatchId: matched.length > 0 ? matched[0] : null,
   748→   748→    });
   749→   749→    return matched;
   750→   750→  },
   751→   751→
   752→   752→  replaceInNode: (nodeId, search, replacement, opts) => {
   753→   753→    const caseSensitive = opts?.caseSensitive ?? false;
   754→   754→    if (!search) return 0;
   755→   755→    const node = get().nodes.find((n) => n.id === nodeId);
   756→   756→    if (!node) return 0;
   757→   757→    // Build a case-sensitive or case-insensitive regex; escape regex special chars.
   758→   758→    const flags = caseSensitive ? "g" : "gi";
   759→   759→    const escaped = search.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   760→   760→    const re = new RegExp(escaped, flags);
   761→   761→    let count = 0;
   762→   762→    const patch: Partial<MapNode> = {};
   763→   763→    if (node.title) {
   764→   764→      const matches = node.title.match(re);
   765→   765→      if (matches && matches.length > 0) {
   766→   766→        patch.title = node.title.replace(re, replacement);
   767→   767→        count += matches.length;
   768→   768→      }
   769→   769→    }
   770→   770→    if (node.content) {
   771→   771→      const matches = node.content.match(re);
   772→   772→      if (matches && matches.length > 0) {
   773→   773→        patch.content = node.content.replace(re, replacement);
   774→   774→        count += matches.length;
   775→   775→      }
   776→   776→    }
   777→   777→    if (count > 0) {
   778→   778→      get().pushHistory();
   779→   779→      get().updateNode(nodeId, patch);
   780→   780→    }
   781→   781→    return count;
   782→   782→  },
   783→   783→
   784→   784→  replaceAll: (search, replacement, opts) => {
   785→   785→    const caseSensitive = opts?.caseSensitive ?? false;
   786→   786→    if (!search) return 0;
   787→   787→    const matchedIds = get().searchMatches;
   788→   788→    if (matchedIds.length === 0) return 0;
   789→   789→    const flags = caseSensitive ? "g" : "gi";
   790→   790→    const escaped = search.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   791→   791→    const re = new RegExp(escaped, flags);
   792→   792→    let total = 0;
   793→   793→    const nodes = get().nodes;
   794→   794→    const updates = new Map<string, Partial<MapNode>>();
   795→   795→    for (const id of matchedIds) {
   796→   796→      const n = nodes.find((x) => x.id === id);
   797→   797→      if (!n) continue;
   798→   798→      const patch: Partial<MapNode> = {};
   799→   799→      if (n.title) {
   800→   800→        const m = n.title.match(re);
   801→   801→        if (m && m.length > 0) {
   802→   802→          patch.title = n.title.replace(re, replacement);
   803→   803→          total += m.length;
   804→   804→        }
   805→   805→      }
   806→   806→      if (n.content) {
   807→   807→        const m = n.content.match(re);
   808→   808→        if (m && m.length > 0) {
   809→   809→          patch.content = n.content.replace(re, replacement);
   810→   810→          total += m.length;
   811→   811→        }
   812→   812→      }
   813→   813→      if (Object.keys(patch).length > 0) {
   814→   814→        updates.set(id, patch);
   815→   815→      }
   816→   816→    }
   817→   817→    if (updates.size > 0) {
   818→   818→      get().pushHistory();
   819→   819→      set((s) => ({
   820→   820→        nodes: s.nodes.map((n) =>
   821→   821→          updates.has(n.id)
   822→   822→            ? { ...n, ...updates.get(n.id)!, updatedAt: new Date().toISOString() }
   823→   823→            : n
   824→   824→        ),
   825→   825→        dirty: true,
   826→   826→      }));
   827→   827→    }
   828→   828→    return total;
   829→   829→  },
   830→   830→}));

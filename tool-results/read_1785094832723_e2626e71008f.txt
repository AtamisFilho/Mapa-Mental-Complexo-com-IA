    21→    21→// ── Alignment guides (Task 14-C) ────────────────────────────────────────
    22→    22→// A guide is a dashed line drawn across the viewport when the dragged
    23→    23→// node's center / edge snaps into alignment with another visible node.
    24→    24→interface AlignmentGuide {
    25→    25→  // "horizontal" → line drawn horizontally at y=position (centers align on Y)
    26→    26→  // "vertical"   → line drawn vertically   at x=position (centers align on X)
    27→    27→  type: "horizontal" | "vertical";
    28→    28→  position: number;
    29→    29→  // World coords for the distance badge between the dragged node and the matched sibling
    30→    30→  badgeX: number;
    31→    31→  badgeY: number;
    32→    32→  distance: number; // px (rounded)
    33→    33→}
    34→    34→
    35→    35→interface AlignmentResult {
    36→    36→  guides: AlignmentGuide[];
    37→    37→  snapX: number | null; // absolute world X to snap the dragged node to (top-left)
    38→    38→  snapY: number | null; // absolute world Y to snap the dragged node to (top-left)
    39→    39→}
    40→    40→
    41→    41→// Tolerance (in world px) within which an alignment is detected.
    42→    42→const ALIGNMENT_TOLERANCE = 6;
    43→    43→// When the visible node count exceeds this, only nodes within CULL_RADIUS of
    44→    44→// the dragged node are compared — keeps the comparison O(nearby) on huge maps.
    45→    45→const ALIGNMENT_MAX_NODES = 100;
    46→    46→const ALIGNMENT_CULL_RADIUS = 1800;
    47→    47→// How far (in world px) the guide line extends on either side of the viewport.
    48→    48→// The SVG container is clipped by the canvas, so a very large extent is fine.
    49→    49→const GUIDE_EXTENT = 100000;
    50→    50→
    51→    51→/**
    52→    52→ * Compute alignment guides for the dragged node against every other visible
    53→    53→ * node. Also returns the snap target (top-left X / Y of the dragged node) so
    54→    54→ * the caller can apply a magnetic snap when a guide is active.
    55→    55→ */
    56→    56→function computeAlignmentGuides(
    57→    57→  draggedId: string,
    58→    58→  draggedX: number,
    59→    59→  draggedY: number,
    60→    60→  draggedW: number,
    61→    61→  draggedH: number,
    62→    62→  allNodes: { id: string; x: number; y: number; width: number; height: number }[]
    63→    63→): AlignmentResult {
    64→    64→  const draggedCx = draggedX + draggedW / 2;
    65→    65→  const draggedCy = draggedY + draggedH / 2;
    66→    66→  const draggedRight = draggedX + draggedW;
    67→    67→  const draggedBottom = draggedY + draggedH;
    68→    68→
    69→    69→  const guides: AlignmentGuide[] = [];
    70→    70→  let snapX: number | null = null;
    71→    71→  let snapY: number | null = null;
    72→    72→
    73→    73→  // Cull to nearby nodes when there are many visible nodes (perf cap).
    74→    74→  let candidates = allNodes.filter((n) => n.id !== draggedId);
    75→    75→  if (candidates.length > ALIGNMENT_MAX_NODES) {
    76→    76→    candidates = candidates.filter((n) => {
    77→    77→      const dx = Math.abs(n.x + n.width / 2 - draggedCx);
    78→    78→      const dy = Math.abs(n.y + n.height / 2 - draggedCy);
    79→    79→      return dx < ALIGNMENT_CULL_RADIUS && dy < ALIGNMENT_CULL_RADIUS;
    80→    80→    });
    81→    81→  }
    82→    82→
    83→    83→  for (const other of candidates) {
    84→    84→    const oW = other.width;
    85→    85→    const oH = other.height;
    86→    86→    const oCx = other.x + oW / 2;
    87→    87→    const oCy = other.y + oH / 2;
    88→    88→    const oRight = other.x + oW;
    89→    89→    const oBottom = other.y + oH;
    90→    90→
    91→    91→    // ── Vertical guide (alignment on the X axis) ───────────────────────
    92→    92→    // Possible matches: center-X, left edge, right edge.
    93→    93→    let vertPos: number | null = null;
    94→    94→    if (Math.abs(draggedCx - oCx) < ALIGNMENT_TOLERANCE) {
    95→    95→      vertPos = oCx;
    96→    96→      if (snapX === null) snapX = oCx - draggedW / 2;
    97→    97→    } else if (Math.abs(draggedX - other.x) < ALIGNMENT_TOLERANCE) {
    98→    98→      vertPos = other.x;
    99→    99→      if (snapX === null) snapX = other.x;
   100→   100→    } else if (Math.abs(draggedRight - oRight) < ALIGNMENT_TOLERANCE) {
   101→   101→      vertPos = oRight;
   102→   102→      if (snapX === null) snapX = oRight - draggedW;
   103→   103→    }
   104→   104→    if (vertPos !== null) {
   105→   105→      // Distance along the perpendicular (Y) axis: gap between the two boxes.
   106→   106→      const isDraggedBelow = draggedY > other.y;
   107→   107→      const gap = isDraggedBelow
   108→   108→        ? Math.max(0, draggedY - oBottom)
   109→   109→        : Math.max(0, other.y - draggedBottom);
   110→   110→      const badgeY = isDraggedBelow
   111→   111→        ? (oBottom + draggedY) / 2
   112→   112→        : (draggedBottom + other.y) / 2;
   113→   113→      guides.push({
   114→   114→        type: "vertical",
   115→   115→        position: vertPos,
   116→   116→        badgeX: vertPos,
   117→   117→        badgeY,
   118→   118→        distance: Math.round(gap),
   119→   119→      });
   120→   120→    }
   121→   121→
   122→   122→    // ── Horizontal guide (alignment on the Y axis) ────────────────────
   123→   123→    // Possible matches: center-Y, top edge, bottom edge.
   124→   124→    let horizPos: number | null = null;
   125→   125→    if (Math.abs(draggedCy - oCy) < ALIGNMENT_TOLERANCE) {
   126→   126→      horizPos = oCy;
   127→   127→      if (snapY === null) snapY = oCy - draggedH / 2;
   128→   128→    } else if (Math.abs(draggedY - other.y) < ALIGNMENT_TOLERANCE) {
   129→   129→      horizPos = other.y;
   130→   130→      if (snapY === null) snapY = other.y;
   131→   131→    } else if (Math.abs(draggedBottom - oBottom) < ALIGNMENT_TOLERANCE) {
   132→   132→      horizPos = oBottom;
   133→   133→      if (snapY === null) snapY = oBottom - draggedH;
   134→   134→    }
   135→   135→    if (horizPos !== null) {
   136→   136→      const isDraggedRight = draggedX > other.x;
   137→   137→      const gap = isDraggedRight
   138→   138→        ? Math.max(0, draggedX - oRight)
   139→   139→        : Math.max(0, other.x - draggedRight);
   140→   140→      const badgeX = isDraggedRight
   141→   141→        ? (oRight + draggedX) / 2
   142→   142→        : (draggedRight + other.x) / 2;
   143→   143→      guides.push({
   144→   144→        type: "horizontal",
   145→   145→        position: horizPos,
   146→   146→        badgeX,
   147→   147→        badgeY: horizPos,
   148→   148→        distance: Math.round(gap),
   149→   149→      });
   150→   150→    }
   151→   151→  }
   152→   152→
   153→   153→  return { guides, snapX, snapY };
   154→   154→}

   450→   450→        setBoxSel({ startX: d.startX, startY: d.startY, endX: e.clientX, endY: e.clientY });
   451→   451→      } else if (d.type === "node" && d.nodeId) {
   452→   452→        const worldDx = dx / viewport.zoom;

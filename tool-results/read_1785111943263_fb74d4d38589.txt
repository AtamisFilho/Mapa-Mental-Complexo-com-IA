    22→// ── Alignment guides (Task 14-C) ────────────────────────────────────────
    23→// A guide is a dashed line drawn across the viewport when the dragged
    24→// node's center / edge snaps into alignment with another visible node.
    25→interface AlignmentGuide {
    26→  // "horizontal" → line drawn horizontally at y=position (centers align on Y)
    27→  // "vertical"   → line drawn vertically   at x=position (centers align on X)
    28→  type: "horizontal" | "vertical";
    29→  position: number;
    30→  // World coords for the distance badge between the dragged node and the matched sibling
    31→  badgeX: number;
    32→  badgeY: number;
    33→  distance: number; // px (rounded)
    34→}
    35→
    36→interface AlignmentResult {
    37→  guides: AlignmentGuide[];
    38→  snapX: number | null; // absolute world X to snap the dragged node to (top-left)
    39→  snapY: number | null; // absolute world Y to snap the dragged node to (top-left)
    40→}
    41→
    42→// Tolerance (in world px) within which an alignment is detected.
    43→const ALIGNMENT_TOLERANCE = 6;
    44→// When the visible node count exceeds this, only nodes within CULL_RADIUS of
    45→// the dragged node are compared — keeps the comparison O(nearby) on huge maps.
    46→const ALIGNMENT_MAX_NODES = 100;
    47→const ALIGNMENT_CULL_RADIUS = 1800;
    48→// How far (in world px) the guide line extends on either side of the viewport.
    49→// The SVG container is clipped by the canvas, so a very large extent is fine.
    50→const GUIDE_EXTENT = 100000;
    51→
    52→/**
    53→ * Compute alignment guides for the dragged node against every other visible
    54→ * node. Also returns the snap target (top-left X / Y of the dragged node) so
    55→ * the caller can apply a magnetic snap when a guide is active.
    56→ */
    57→function computeAlignmentGuides(
    58→  draggedId: string,
    59→  draggedX: number,
    60→  draggedY: number,
    61→  draggedW: number,
    62→  draggedH: number,
    63→  allNodes: { id: string; x: number; y: number; width: number; height: number }[]
    64→): AlignmentResult {
    65→  const draggedCx = draggedX + draggedW / 2;
    66→  const draggedCy = draggedY + draggedH / 2;
    67→  const draggedRight = draggedX + draggedW;
    68→  const draggedBottom = draggedY + draggedH;
    69→
    70→  const guides: AlignmentGuide[] = [];
    71→  let snapX: number | null = null;
    72→  let snapY: number | null = null;
    73→
    74→  // Cull to nearby nodes when there are many visible nodes (perf cap).
    75→  let candidates = allNodes.filter((n) => n.id !== draggedId);
    76→  if (candidates.length > ALIGNMENT_MAX_NODES) {
    77→    candidates = candidates.filter((n) => {
    78→      const dx = Math.abs(n.x + n.width / 2 - draggedCx);
    79→      const dy = Math.abs(n.y + n.height / 2 - draggedCy);
    80→      return dx < ALIGNMENT_CULL_RADIUS && dy < ALIGNMENT_CULL_RADIUS;
    81→    });
    82→  }
    83→
    84→  for (const other of candidates) {
    85→    const oW = other.width;
    86→    const oH = other.height;
    87→    const oCx = other.x + oW / 2;
    88→    const oCy = other.y + oH / 2;
    89→    const oRight = other.x + oW;
    90→    const oBottom = other.y + oH;
    91→
    92→    // ── Vertical guide (alignment on the X axis) ───────────────────────
    93→    // Possible matches: center-X, left edge, right edge.
    94→    let vertPos: number | null = null;
    95→    if (Math.abs(draggedCx - oCx) < ALIGNMENT_TOLERANCE) {
    96→      vertPos = oCx;
    97→      if (snapX === null) snapX = oCx - draggedW / 2;
    98→    } else if (Math.abs(draggedX - other.x) < ALIGNMENT_TOLERANCE) {
    99→      vertPos = other.x;
   100→      if (snapX === null) snapX = other.x;
   101→    } else if (Math.abs(draggedRight - oRight) < ALIGNMENT_TOLERANCE) {
   102→      vertPos = oRight;
   103→      if (snapX === null) snapX = oRight - draggedW;
   104→    }
   105→    if (vertPos !== null) {
   106→      // Distance along the perpendicular (Y) axis: gap between the two boxes.
   107→      const isDraggedBelow = draggedY > other.y;
   108→      const gap = isDraggedBelow
   109→        ? Math.max(0, draggedY - oBottom)
   110→        : Math.max(0, other.y - draggedBottom);
   111→      const badgeY = isDraggedBelow
   112→        ? (oBottom + draggedY) / 2
   113→        : (draggedBottom + other.y) / 2;
   114→      guides.push({
   115→        type: "vertical",
   116→        position: vertPos,
   117→        badgeX: vertPos,
   118→        badgeY,
   119→        distance: Math.round(gap),
   120→      });
   121→    }
   122→
   123→    // ── Horizontal guide (alignment on the Y axis) ────────────────────
   124→    // Possible matches: center-Y, top edge, bottom edge.
   125→    let horizPos: number | null = null;
   126→    if (Math.abs(draggedCy - oCy) < ALIGNMENT_TOLERANCE) {
   127→      horizPos = oCy;
   128→      if (snapY === null) snapY = oCy - draggedH / 2;
   129→    } else if (Math.abs(draggedY - other.y) < ALIGNMENT_TOLERANCE) {
   130→      horizPos = other.y;
   131→      if (snapY === null) snapY = other.y;
   132→    } else if (Math.abs(draggedBottom - oBottom) < ALIGNMENT_TOLERANCE) {
   133→      horizPos = oBottom;
   134→      if (snapY === null) snapY = oBottom - draggedH;
   135→    }
   136→    if (horizPos !== null) {
   137→      const isDraggedRight = draggedX > other.x;
   138→      const gap = isDraggedRight
   139→        ? Math.max(0, draggedX - oRight)
   140→        : Math.max(0, other.x - draggedRight);
   141→      const badgeX = isDraggedRight
   142→        ? (oRight + draggedX) / 2
   143→        : (draggedRight + other.x) / 2;
   144→      guides.push({
   145→        type: "horizontal",
   146→        position: horizPos,
   147→        badgeX,
   148→        badgeY: horizPos,
   149→        distance: Math.round(gap),
   150→      });
   151→    }
   152→  }
   153→
   154→  return { guides, snapX, snapY };
   155→}

   418→  const visibleNodeIds = useMemo(() => {
   419→    const collapsedIds = new Set(nodes.filter((n) => n.collapsed).map((n) => n.id));
   420→    if (collapsedIds.size === 0) return null; // all visible
   421→    const hidden = new Set<string>();
   422→    // Build child map
   423→    const childrenOf = new Map<string, string[]>();
   424→    for (const e of edges) {
   425→      if (!childrenOf.has(e.sourceId)) childrenOf.set(e.sourceId, []);
   426→      childrenOf.get(e.sourceId)!.push(e.targetId);
   427→    }
   428→    // BFS from each collapsed node
   429→    const queue = [...collapsedIds];
   430→    while (queue.length) {
   431→      const cur = queue.shift()!;
   432→      for (const child of childrenOf.get(cur) ?? []) {
   433→        if (!hidden.has(child)) {
   434→          hidden.add(child);
   435→          queue.push(child);
   436→        }
   437→      }
   438→    }
   439→    return hidden;
   440→  }, [nodes, edges]);
   441→

   467→        // Alignment guides: detect center / edge alignments with sibling nodes
   468→        // and (a) render dashed guide lines + distance badges, (b) magnetically
   469→        // snap the dragged node to the exact aligned position.
   470→        if (alignmentGuidesEnabled) {

   472→          if (draggedNode) {
   473→            const w = draggedNode.width;
   474→            const h = draggedNode.height;
   475→            const result = computeAlignmentGuides(
   476→              d.nodeId,
   477→              newX,
   478→              newY,
   479→              w,
   480→              h,
   481→              visibleNodes
   482→            );
   483→            // Only update state if the guide set actually changed — avoids
   484→            // unnecessary re-renders on every pointer-move tick.
   485→            setActiveGuides((prev) => {
   486→              if (
   487→                prev.length === result.guides.length &&
   488→                prev.every(
   489→                  (g, i) =>
   490→                    g.type === result.guides[i].type &&
   491→                    g.position === result.guides[i].position &&
   492→                    g.badgeX === result.guides[i].badgeX &&
   493→                    g.badgeY === result.guides[i].badgeY &&
   494→                    g.distance === result.guides[i].distance
   495→                )
   496→              ) {
   497→                return prev;
   498→              }
   499→              return result.guides;
   500→            });
   501→            if (result.snapX !== null) newX = result.snapX;
   502→            if (result.snapY !== null) newY = result.snapY;
   503→          } else if (activeGuides.length > 0) {
   504→            setActiveGuides([]);
   505→          }
   506→        } else if (activeGuides.length > 0) {
   507→          setActiveGuides([]);
   508→        }
   509→
   510→        updateNode(d.nodeId, { x: newX, y: newY });
   511→
   512→        // ── Reparent detection (Task 16-B) ────────────────────────────────
   513→        // Check if the dragged node's center overlaps with another node's
   514→        // bounding box. If so, that node becomes the reparent target.
   515→        const draggedNodeCurrent = visibleNodes.find((n) => n.id === d.nodeId);
   516→        if (draggedNodeCurrent) {
   517→          const draggedCx = newX + draggedNodeCurrent.width / 2;
   518→          const draggedCy = newY + draggedNodeCurrent.height / 2;
   519→
   520→          let newTarget: string | null = null;
   521→          for (const other of visibleNodes) {
   522→            if (other.id === d.nodeId) continue; // skip self
   523→            // Check if dragged center is inside other node's bounding box
   524→            if (
   525→              draggedCx >= other.x &&
   526→              draggedCx <= other.x + other.width &&
   527→              draggedCy >= other.y &&
   528→              draggedCy <= other.y + other.height
   529→            ) {
   530→              // Validate: cannot reparent to own descendant
   531→              if (!isDescendantOf(other.id, d.nodeId)) {
   532→                newTarget = other.id;
   533→              }
   534→              break; // only one target at a time
   535→            }
   536→          }
   537→          setReparentTarget(newTarget);
   538→        }
   539→      }
   540→    },
   541→    [isDragging, connectingFrom, viewport.zoom, panBy, updateNode, snap, screenToWorld, setCursorWorld, alignmentGuidesEnabled, activeGuides.length, visibleNodes, setReparentTarget, isDescendantOf]
   542→  );
   543→
   544→  // Pointer up
   545→  const handlePointerUp = useCallback(() => {
   546→    const d = dragRef.current;
   547→    // ── Reparent execution (Task 16-B) ──────────────────────────────────
   548→    if (d?.type === "node" && d.nodeId && reparentTargetId) {
   549→      const success = reparentNode(d.nodeId, reparentTargetId);
   550→      if (success) {
   551→        const targetNode = nodes.find((n) => n.id === reparentTargetId);
   552→        toast({
   553→          title: `Nó movido para ${targetNode?.title ?? "nó"}`,
   554→          variant: "success",
   555→        });
   556→      }
   557→      setReparentTarget(null);
   558→      setDraggedNode(null);
   559→    } else {
   560→      setReparentTarget(null);
   561→      setDraggedNode(null);
   562→    }
   563→
   564→    if (d?.type === "box" && boxSel) {
   565→      // Convert box to world coords, then select nodes within
   566→      const rect = containerRef.current?.getBoundingClientRect();
   567→      if (rect) {
   568→        const x1 = Math.min(boxSel.startX, boxSel.endX) - rect.left;
   569→        const y1 = Math.min(boxSel.startY, boxSel.endY) - rect.top;
   570→        const x2 = Math.max(boxSel.startX, boxSel.endX) - rect.left;
   571→        const y2 = Math.max(boxSel.startY, boxSel.endY) - rect.top;
   572→        // World-space rect
   573→        const wx1 = (x1 - viewport.x) / viewport.zoom;
   574→        const wy1 = (y1 - viewport.y) / viewport.zoom;
   575→        const wx2 = (x2 - viewport.x) / viewport.zoom;
   576→        const wy2 = (y2 - viewport.y) / viewport.zoom;
   577→        const hits = nodes.filter(
   578→          (n) => n.x + n.width >= wx1 && n.x <= wx2 && n.y + n.height >= wy1 && n.y <= wy2
   579→        );
   580→        // Box-select only when meaningful drag (>=4px), else treat as click → clear
   581→        const movedEnough = Math.abs(boxSel.endX - boxSel.startX) > 4 || Math.abs(boxSel.endY - boxSel.startY) > 4;
   582→        if (movedEnough) {
   583→          if (hits.length > 0) {
   584→            // Replace selection with hits
   585→            useMindMapStore.setState({ selectedNodeIds: hits.map((n) => n.id), selectedEdgeIds: [] });
   586→          } else {
   587→            clearSelection();
   588→          }
   589→        } else {
   590→          clearSelection();
   591→        }
   592→      }
   593→      setBoxSel(null);
   594→    }
   595→    dragRef.current = null;
   596→    setIsDragging(false);
   597→    setDragType(null);
   598→    // Always clear alignment guides on pointer-up — they should only appear
   599→    // while a node is being actively dragged.
   600→    setActiveGuides([]);
   601→  }, [boxSel, viewport, nodes, clearSelection, reparentTargetId, reparentNode, setReparentTarget, setDraggedNode, toast]);
   602→
   603→  // Wheel zoom
   604→  const handleWheel = useCallback(
   605→    (e: React.WheelEvent) => {
   606→      e.preventDefault();
   607→      const factor = e.deltaY < 0 ? 1.08 : 0.92;
   608→      const rect = containerRef.current?.getBoundingClientRect();
   609→      if (!rect) return;
   610→      const cx = e.clientX - rect.left;
   611→      const cy = e.clientY - rect.top;
   612→      zoomBy(factor, cx, cy);
   613→    },
   614→    [zoomBy]
   615→  );
   616→
   617→  // Double click on canvas — add node. Double click on node → open editor.
   618→  const handleDoubleClick = useCallback(
   619→    (e: React.MouseEvent) => {
   620→      const target = e.target as HTMLElement;
   621→      const isCanvasBg = target === containerRef.current || !!target.dataset?.canvas;
   622→      if (!isCanvasBg) {
   623→        // Was a node double-click? Look for data-node-id
   624→        const nodeEl = target.closest("[data-node-id]") as HTMLElement | null;
   625→        if (nodeEl?.dataset.nodeId) {
   626→          // Select and open editor
   627→          selectNodes(nodeEl.dataset.nodeId, false);
   628→          onOpenNodeEditor();
   629→          return;
   630→        }
   631→        return;
   632→      }
   633→      const world = screenToWorld(e.clientX, e.clientY);
   634→      const id = addNode({
   635→        title: "Novo conceito",
   636→        x: snap(world.x - 100),
   637→        y: snap(world.y - 40),
   638→        kind: "concept",
   639→        width: 220,
   640→        height: 88,
   641→      });
   642→      focusNode(id);
   643→      onOpenNodeEditor();
   644→    },
   645→    [screenToWorld, addNode, snap, focusNode, onOpenNodeEditor, selectNodes]
   646→  );
   647→
   648→  // Keyboard shortcuts
   649→  useEffect(() => {
   650→    const handler = (e: KeyboardEvent) => {
   651→      // Skip if user is typing in an input/textarea/contenteditable
   652→      const target = e.target as HTMLElement;
   653→      if (target && (target.tagName === "INPUT" || target.tagName === "TEXTAREA" || target.isContentEditable)) {
   654→        return;
   655→      }
   656→      // Duplicate node: Ctrl/Cmd+D
   657→      if ((e.ctrlKey || e.metaKey) && (e.key === "d" || e.key === "D")) {
   658→        if (selectedNodeIds.length > 0) {
   659→          e.preventDefault();
   660→          pushHistory();
   661→          // Duplicate the first selected node
   662→          duplicateNode(selectedNodeIds[0]);
   663→        }
   664→        return;
   665→      }
   666→      if ((e.ctrlKey || e.metaKey) && e.key === "z") {
   667→        if (!undoRedo) return;
   668→        e.preventDefault();
   669→        undo();
   670→        return;
   671→      }
   672→      if ((e.ctrlKey || e.metaKey) && (e.key === "y" || (e.shiftKey && e.key === "z"))) {
   673→        if (!undoRedo) return;
   674→        e.preventDefault();
   675→        redo();
   676→        return;
   677→      }
   678→      if (e.key === "Delete" || e.key === "Backspace") {
   679→        if (selectedNodeIds.length > 0 && !confirmDelete) {
   680→          for (const id of selectedNodeIds) {
   681→            deleteNode(id);
   682→          }
   683→        }
   684→        return;
   685→      }
   686→      if (e.key === "Escape") {
   687→        clearSelection();
   688→        setConnectingFrom(null);
   689→        return;
   690→      }
   691→      if (!shortcutsEnabled) return;
   692→      // Fit to view
   693→      if (e.key === "f" || e.key === "F") {
   694→        e.preventDefault();
   695→        fitToView(80);
   696→        return;
   697→      }
   698→      // Add node shortcuts: C/P/A/I/R/O
   699→      const keyMap: Record<string, NodeKind> = {
   700→        c: "concept", p: "question", a: "action", i: "idea", r: "resource", o: "goal",
   701→      };
   702→      const k = e.key.toLowerCase();
   703→      if (keyMap[k]) {
   704→        e.preventDefault();
   705→        const cx = window.innerWidth / 2;
   706→        const cy = window.innerHeight / 2;
   707→        const wx = (cx - viewport.x) / viewport.zoom - 110;
   708→        const wy = (cy - viewport.y) / viewport.zoom - 44;
   709→        addNode({ title: "Novo " + NODE_KIND_META[keyMap[k]].label, kind: keyMap[k], x: wx, y: wy, width: 220, height: 88 });
   710→        return;
   711→      }
   712→      // E or Enter — open Node Editor for the selected node
   713→      if (k === "e" || e.key === "Enter") {
   714→        if (selectedNodeIds.length > 0) {
   715→          e.preventDefault();
   716→          onOpenNodeEditor();
   717→        }
   718→        return;
   719→      }
   720→    };
   721→    window.addEventListener("keydown", handler);
   722→    return () => window.removeEventListener("keydown", handler);
   723→  }, [undoRedo, undo, redo, selectedNodeIds, confirmDelete, deleteNode, duplicateNode, pushHistory, clearSelection, setConnectingFrom, shortcutsEnabled, fitToView, viewport, addNode, onOpenNodeEditor]);
   724→
   725→  const canvasBackground = useSettingsStore((s) => s.settings.visual.canvasBackground);
   726→  const bgClass = canvasBackground === "grid" ? "canvas-grid-bg"
   727→    : canvasBackground === "gradient" ? "canvas-gradient-bg"
   728→    : canvasBackground === "dots" ? "canvas-dots-bg"
   729→    : "canvas-plain-bg";
   730→
   731→  const visibleEdges = visibleNodeIds
   732→    ? edges.filter((e) => !visibleNodeIds.has(e.sourceId) && !visibleNodeIds.has(e.targetId))
   733→    : edges;
   734→
   735→  // Compute highlighted nodes (ancestors + descendants of selected nodes)
   736→  const highlightedNodeIds = useMemo(() => {
   737→    if (selectedNodeIds.length === 0) return new Set<string>();
   738→    const highlighted = new Set<string>();
   739→    // Build parent map and child map from edges
   740→    const parentOf = new Map<string, string>(); // child → parent
   741→    const childrenOf = new Map<string, string[]>(); // parent → children
   742→    for (const e of edges) {
   743→      parentOf.set(e.targetId, e.sourceId);
   744→      if (!childrenOf.has(e.sourceId)) childrenOf.set(e.sourceId, []);
   745→      childrenOf.get(e.sourceId)!.push(e.targetId);
   746→    }
   747→    // For each selected node, trace ancestors upward
   748→    for (const selId of selectedNodeIds) {
   749→      let current = parentOf.get(selId);
   750→      while (current) {
   751→        if (!selectedNodeIds.includes(current)) highlighted.add(current);
   752→        current = parentOf.get(current);
   753→      }
   754→    }
   755→    // For each selected node, trace descendants downward
   756→    const queue = [...selectedNodeIds];
   757→    while (queue.length) {
   758→      const cur = queue.shift()!;
   759→      for (const child of childrenOf.get(cur) ?? []) {
   760→        if (!selectedNodeIds.includes(child) && !highlighted.has(child)) {
   761→          highlighted.add(child);
   762→          queue.push(child);
   763→        }
   764→      }
   765→    }
   766→    return highlighted;
   767→  }, [selectedNodeIds, edges]);
   768→
   769→  return (
   770→    <div
   771→      ref={containerRef}
   772→      data-canvas="true"
   773→      className={`relative flex-1 overflow-hidden ${bgClass}`}
   774→      style={{
   775→        cursor:
   776→          isDragging && dragType === "pan"
   777→            ? "grabbing"
   778→            : tool === "pan"
   779→              ? "grab"
   780→              : tool === "connect"
   781→                ? "crosshair"
   782→                : isDragging
   783→                  ? "grabbing"
   784→                  : "default",
   785→      }}
   786→      onPointerDown={handleCanvasPointerDown}
   787→      onPointerMove={handlePointerMove}
   788→      onPointerUp={handlePointerUp}
   789→      onWheel={handleWheel}
   790→      onDoubleClick={handleDoubleClick}
   791→      onContextMenu={(e) => e.preventDefault()}
   792→    >
   793→      {/* transform layer */}
   794→      <div
   795→        data-canvas="true"
   796→        style={{
   797→          transform: `translate(${viewport.x}px, ${viewport.y}px) scale(${viewport.zoom})`,
   798→          transformOrigin: "0 0",
   799→          position: "absolute",
   800→          left: 0,
   801→          top: 0,
   802→          willChange: "transform",
   803→        }}
   804→      >
   805→        <MapEdges
   806→          nodes={visibleNodes}
   807→          edges={visibleEdges}
   808→          connectingFrom={connectingFrom}
   809→          cursorWorld={cursorWorld}
   810→        />
   811→        {/* Alignment / snap guides — rendered ABOVE edges but BELOW nodes
   812→            (z-order in SVG/HTML is determined by document order, so this
   813→             layer sits between MapEdges and the node motion.divs). Only
   814→             rendered while a node is being actively dragged. */}
   815→        {activeGuides.length > 0 && (
   816→          <svg
   817→            style={{
   818→              position: "absolute",
   819→              left: 0,
   820→              top: 0,
   821→              width: 1,
   822→              height: 1,
   823→              overflow: "visible",
   824→              pointerEvents: "none",
   825→              shapeRendering: "crispEdges",
   826→            }}
   827→            aria-hidden="true"
   828→          >
   829→            {activeGuides.map((g, i) => {
   830→              const badgeW = 38;
   831→              const badgeH = 18;
   832→              return (
   833→                <g key={`guide-${i}`}>
   834→                  {g.type === "horizontal" ? (
   835→                    <line
   836→                      className="alignment-guide"
   837→                      x1={-GUIDE_EXTENT}
   838→                      y1={g.position}
   839→                      x2={GUIDE_EXTENT}
   840→                      y2={g.position}
   841→                      stroke="#ec4899"
   842→                      strokeWidth={1}
   843→                      strokeDasharray="4 4"
   844→                      vectorEffect="non-scaling-stroke"
   845→                    />
   846→                  ) : (
   847→                    <line
   848→                      className="alignment-guide"
   849→                      x1={g.position}
   850→                      y1={-GUIDE_EXTENT}
   851→                      x2={g.position}
   852→                      y2={GUIDE_EXTENT}
   853→                      stroke="#ec4899"
   854→                      strokeWidth={1}
   855→                      strokeDasharray="4 4"
   856→                      vectorEffect="non-scaling-stroke"
   857→                    />
   858→                  )}
   859→                  {/* Distance badge between the dragged node and the matched sibling */}
   860→                  <rect
   861→                    className="alignment-guide-badge-bg"
   862→                    x={g.badgeX - badgeW / 2}
   863→                    y={g.badgeY - badgeH / 2}
   864→                    width={badgeW}
   865→                    height={badgeH}
   866→                    rx={4}
   867→                  />
   868→                  <text
   869→                    className="alignment-guide-badge"
   870→                    x={g.badgeX}
   871→                    y={g.badgeY + 3.5}
   872→                    textAnchor="middle"
   873→                  >
   874→                    {g.distance}px
   875→                  </text>
   876→                </g>
   877→              );
   878→            })}
   879→          </svg>
   880→        )}
   881→        <AnimatePresence>
   882→          {visibleNodes.map((node) => (
   883→            <MapNodeView
   884→              key={node.id}
   885→              node={node}
   886→              onPointerDown={handleNodePointerDown}
   887→              onConnectHandle={handleConnectHandle}
   888→              onContextMenu={handleNodeContextMenu}
   889→              isHighlighted={highlightedNodeIds.has(node.id)}
   890→              isReparentTarget={reparentTargetId === node.id}
   891→              isBeingDraggedForReparent={draggedNodeId === node.id}
   892→            />
   893→          ))}
   894→        </AnimatePresence>
   895→      </div>
   896→
   897→      {/* Empty state hint */}
   898→      {nodes.length === 0 && !isDragging && (
   899→        <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
   900→          <div className="text-center max-w-md px-6">
   901→            <div className="mx-auto mb-4 h-16 w-16 rounded-2xl bg-primary/10 flex items-center justify-center relative">
   902→              <Sparkles className="h-8 w-8 text-primary" />
   903→              <div className="absolute inset-0 rounded-2xl bg-primary/10 animate-ping opacity-30" />
   904→            </div>
   905→            <p className="text-lg font-semibold brand-gradient mb-1">Comece seu mapa mental</p>
   906→            <p className="text-sm text-muted-foreground mb-3">
   907→              Clique duplo no canvas para adicionar um nó, ou use o botão <strong className="text-foreground">Adicionar</strong> na barra.
   908→            </p>
   909→            <div className="flex flex-wrap gap-1.5 justify-center text-[10px] text-muted-foreground mb-3">
   910→              <span className="flex items-center gap-1"><kbd className="px-1.5 py-0.5 rounded border border-border bg-muted">C</kbd> Conceito</span>
   911→              <span className="flex items-center gap-1"><kbd className="px-1.5 py-0.5 rounded border border-border bg-muted">P</kbd> Pergunta</span>
   912→              <span className="flex items-center gap-1"><kbd className="px-1.5 py-0.5 rounded border border-border bg-muted">A</kbd> Ação</span>
   913→              <span className="flex items-center gap-1"><kbd className="px-1.5 py-0.5 rounded border border-border bg-muted">I</kbd> Ideia</span>
   914→              <span className="flex items-center gap-1"><kbd className="px-1.5 py-0.5 rounded border border-border bg-muted">F</kbd> Ajustar</span>
   915→              <span className="flex items-center gap-1"><kbd className="px-1.5 py-0.5 rounded border border-border bg-muted">⌘K</kbd> Buscar</span>
   916→            </div>
   917→            <p className="text-[11px] text-muted-foreground/70 italic">
   918→              Dica: explore os <strong className="text-foreground/80">templates</strong> no painel Mapas (botão no canto esquerdo).
   919→            </p>
   920→          </div>
   921→        </div>
   922→      )}
   923→
   924→      {/* Connection mode indicator */}
   925→      {tool === "connect" && !connectingFrom && (
   926→        <div className="absolute top-3 left-1/2 -translate-x-1/2 pointer-events-none">
   927→          <div className="bg-primary text-primary-foreground px-3 py-1.5 rounded-full text-xs font-medium shadow-lg fade-in">
   928→            Clique em um nó para iniciar a conexão
   929→          </div>
   930→        </div>
   931→      )}
   932→      {connectingFrom && (
   933→        <div className="absolute top-3 left-1/2 -translate-x-1/2 pointer-events-none">
   934→          <div className="bg-primary text-primary-foreground px-3 py-1.5 rounded-full text-xs font-medium shadow-lg fade-in">
   935→            Clique em outro nó para completar a conexão · Esc para cancelar
   936→          </div>
   937→        </div>
   938→      )}
   939→
   940→      {/* Box-selection visual */}
   941→      {boxSel && (
   942→        <div
   943→          className="absolute pointer-events-none border-2 border-primary/60 bg-primary/15 rounded-sm z-40"
   944→          style={{
   945→            left: Math.min(boxSel.startX, boxSel.endX),
   946→            top: Math.min(boxSel.startY, boxSel.endY),
   947→            width: Math.abs(boxSel.endX - boxSel.startX),
   948→            height: Math.abs(boxSel.endY - boxSel.startY),
   949→          }}
   950→        />
   951→      )}
   952→
   953→      {/* Selection info badge when multiple nodes selected */}
   954→      {selectedNodeIds.length > 1 && !isDragging && (
   955→        <div className="absolute bottom-3 left-1/2 -translate-x-1/2 pointer-events-none">
   956→          <div className="bg-card/95 border border-border text-foreground px-3 py-1.5 rounded-full text-xs font-medium shadow-lg fade-in flex items-center gap-1.5">
   957→            <span className="h-1.5 w-1.5 rounded-full bg-primary" />
   958→            {selectedNodeIds.length} nós selecionados · <kbd className="text-[10px] bg-muted px-1 py-0.5 rounded border border-border">Del</kbd> excluir · <kbd className="text-[10px] bg-muted px-1 py-0.5 rounded border border-border">Ctrl+D</kbd> duplicar 1º
   959→          </div>
   960→        </div>
   961→      )}
   962→
   963→      {/* Node context menu */}
   964→      <NodeContextMenu
   965→        key={contextMenu?.nodeId ?? "none"}
   966→        menuState={contextMenu}
   967→        onClose={() => setContextMenu(null)}
   968→        onEdit={handleContextEdit}
   969→        onExpand={handleContextExpand}
   970→        onDuplicate={handleContextDuplicate}
   971→        onToggleCollapse={handleContextToggleCollapse}
   972→        onConnectFrom={handleContextConnectFrom}
   973→        onColorChange={handleContextColorChange}
   974→        onIconChange={handleContextIconChange}
   975→        onDelete={handleContextDelete}
   976→      />
   977→    </div>
   978→  );
   979→}

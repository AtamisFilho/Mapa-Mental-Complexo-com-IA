    28→// Validate imported JSON structure
    29→function validateImportJSON(data: unknown): { valid: boolean; error?: string } {
    30→  if (!data || typeof data !== "object") {
    31→    return { valid: false, error: "JSON deve ser um objeto." };
    32→  }
    33→  const obj = data as Record<string, unknown>;
    34→  if (!Array.isArray(obj.nodes)) {
    35→    return { valid: false, error: "Campo 'nodes' ausente ou não é um array." };
    36→  }
    37→  if (!Array.isArray(obj.edges)) {
    38→    return { valid: false, error: "Campo 'edges' ausente ou não é um array." };
    39→  }
    40→  for (const n of obj.nodes as Array<Record<string, unknown>>) {
    41→    if (!n.title || typeof n.title !== "string") {
    42→      return { valid: false, error: `Node sem 'title': ${JSON.stringify(n).slice(0, 60)}` };
    43→    }
    44→    if (n.kind && typeof n.kind !== "string") {
    45→      return { valid: false, error: `Node kind inválido: ${n.kind}` };
    46→    }
    47→  }
    48→  for (const e of obj.edges as Array<Record<string, unknown>>) {
    49→    if (!e.sourceId || !e.targetId) {
    50→      return { valid: false, error: `Edge sem 'sourceId' ou 'targetId': ${JSON.stringify(e).slice(0, 60)}` };
    51→    }
    52→  }
    53→  if (obj.nodes.length === 0) {
    54→    return { valid: false, error: "O mapa importado não tem nenhum nó." };
    55→  }
    56→  return { valid: true };
    57→}

   134→        const indent = (depth: number) => "  ".repeat(depth) + "- ";
   135→        const visit = (nodeId: string, depth: number, seen: Set<string>) => {
   136→          if (seen.has(nodeId)) return; // guard against cycles
   137→          seen.add(nodeId);
   138→          const node = nodeMap.get(nodeId);
   139→          if (!node) return;

   141→          md += indent(depth) + `**${node.title}** _(${kindLabel})_`;
   142→          if (node.content) md += ` — ${node.content}`;
   143→          md += "\n";
   144→          if (includeNotes && node.note) {
   145→            md += `${indent(depth + 1)}📝 ${node.note}\n`;
   146→          }
   147→          const kids = children.get(nodeId) ?? [];
   148→          for (const kid of kids) visit(kid, depth + 1, seen);
   149→        };
   150→        for (const root of roots) visit(root.id, 0, new Set());

   159→            treeEdges.add(e.id);
   160→            walk(e.targetId, seen);
   161→          }

   163→      };

   449→        onClose();
   450→      } catch (err) {
   451→        setImportError(`Erro inesperado: ${(err as Error).message}`);
   452→      }

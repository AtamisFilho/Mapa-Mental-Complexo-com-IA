     1→"use client";
     2→
     3→import { useState, useCallback, useRef, useEffect } from "react";
     4→import {
     5→  X,
     6→  Sparkles,
     7→  Expand,
     8→  Map,
     9→  FileText,
    10→  Link2,
    11→  MessageSquare,
    12→  ImagePlus,
    13→  LayoutGrid,
    14→  Loader2,
    15→  ChevronRight,
    16→  BrainCircuit,
    17→  Trash2,
    18→} from "lucide-react";
    19→import { Button } from "@/components/ui/button";
    20→import { Input } from "@/components/ui/input";
    21→import { ScrollArea } from "@/components/ui/scroll-area";
    22→import { useMindMapStore } from "@/store/mindmap-store";
    23→import { useSettingsStore } from "@/store/settings-store";
    24→import { NODE_KIND_META } from "@/lib/settings";
    25→import ReactMarkdown from "react-markdown";
    26→import type { NodeKind } from "@/lib/types";
    27→
    28→type AITab = "expand" | "generate" | "summarize" | "suggest" | "chat" | "image";
    29→
    30→interface Props {
    31→  open: boolean;
    32→  onClose: () => void;
    33→}
    34→
    35→export function AIPanel({ open, onClose }: Props) {
    36→  const [tab, setTab] = useState<AITab>("expand");
    37→  const [loading, loadingAction] = useState<null | string>(null);
    38→  const [result, setResult] = useState<string | null>(null);
    39→  const [topic, setTopic] = useState("");
    40→  const [chatMessages, setChatMessages] = useState<Array<{ role: string; content: string }>>([]);
    41→  const [chatInput, setChatInput] = useState("");
    42→  const chatScrollRef = useRef<HTMLDivElement>(null);
    43→
    44→  const nodes = useMindMapStore((s) => s.nodes);
    45→  const edges = useMindMapStore((s) => s.edges);
    46→  const selectedNodeIds = useMindMapStore((s) => s.selectedNodeIds);
    47→  const addNode = useMindMapStore((s) => s.addNode);
    48→  const addEdge = useMindMapStore((s) => s.addEdge);
    49→  const mergeNodes = useMindMapStore((s) => s.mergeNodes);
    50→  const mergeEdges = useMindMapStore((s) => s.mergeEdges);
    51→  const focusNode = useMindMapStore((s) => s.focusNode);
    52→  const updateNode = useMindMapStore((s) => s.updateNode);
    53→  const mapTitle = useMindMapStore((s) => s.title);
    54→
    55→  const settings = useSettingsStore((s) => s.settings);
    56→
    57→  const selectedNode = nodes.find((n) => n.id === selectedNodeIds[0]);
    58→  const thinking = settings.ai.thinking;
    59→
    60→  // Expand selected node
    61→  const handleExpand = useCallback(async () => {
    62→    if (!selectedNode) return;
    63→    loadingAction("expand");
    64→    setResult(null);
    65→    try {
    66→      const res = await fetch("/api/ai/expand", {
    67→        method: "POST",
    68→        headers: { "Content-Type": "application/json" },
    69→        body: JSON.stringify({
    70→          topic: selectedNode.title,
    71→          context: `${mapTitle}. Nós existentes: ${nodes.slice(0, 20).map((n) => n.title).join(", ")}`,
    72→          count: 5,
    73→          thinking,
    74→        }),
    75→      });
    76→      const data = await res.json();
    77→      if (data.nodes) {
    78→        const children = data.nodes.map((n: { title: string; kind: string; content?: string }, i: number) => {
    79→          const angle = (i / data.nodes.length) * Math.PI * 2 - Math.PI / 2;
    80→          const radius = 280;
    81→          return {
    82→            ...n,
    83→            x: selectedNode.x + selectedNode.width / 2 + Math.cos(angle) * radius - 100,
    84→            y: selectedNode.y + selectedNode.height / 2 + Math.sin(angle) * radius - 40,
    85→            parentId: selectedNode.id,
    86→            width: 200,
    87→            height: 80,
    88→          };
    89→        });
    90→        const newIds: string[] = [];
    91→        for (const child of children) {
    92→          const id = addNode(child);
    93→          newIds.push(id);
    94→        }
    95→        for (const id of newIds) {
    96→          addEdge(selectedNode.id, id);
    97→        }
    98→        setResult(`✅ ${data.nodes.length} conceitos-filho adicionados.`);
    99→      } else {
   100→        setResult(data.error ?? "Erro ao expandir.");
   101→      }
   102→    } catch (e) {
   103→      setResult("Erro de conexão com a IA.");
   104→    }
   105→    loadingAction(null);
   106→  }, [selectedNode, nodes, mapTitle, thinking, addNode, addEdge]);
   107→
   108→  // Generate full map from topic
   109→  const handleGenerate = useCallback(async () => {
   110→    if (!topic.trim()) return;
   111→    loadingAction("generate");
   112→    setResult(null);
   113→    try {
   114→      const res = await fetch("/api/ai/generate", {
   115→        method: "POST",
   116→        headers: { "Content-Type": "application/json" },
   117→        body: JSON.stringify({ topic, depth: 2, breadth: 4, thinking }),
   118→      });
   119→      const data = await res.json();
   120→      if (data.nodes && data.edges) {
   121→        // Load into current map (merge)
   122→        const typedNodes = data.nodes.map((n: Record<string, unknown>) => ({
   123→          ...n,
   124→          id: String(n.id),
   125→          title: String(n.title),
   126→          kind: (["concept", "question", "action", "idea", "resource", "goal"].includes(n.kind as string) ? n.kind : "concept") as NodeKind,
   127→          content: n.content ? String(n.content) : null,
   128→          parentId: n.parentId ? String(n.parentId) : null,
   129→        }));
   130→        const typedEdges = data.edges.map((e: Record<string, unknown>) => ({
   131→          sourceId: String(e.source),
   132→          targetId: String(e.target),
   133→          label: e.label ? String(e.label) : null,
   134→          kind: (["related", "causes", "supports", "contradicts", "depends"].includes(e.kind as string) ? e.kind : "related") as string,
   135→        }));
   136→        mergeNodes(typedNodes);
   137→        for (const e of typedEdges) {
   138→          addEdge(e.sourceId, e.targetId, e.kind, e.label ?? undefined);
   139→        }
   140→        setResult(`✅ Mapa gerado com ${data.nodes.length} nós e ${data.edges.length} conexões.`);
   141→      } else {
   142→        setResult(data.error ?? "Erro ao gerar.");
   143→      }
   144→    } catch (e) {
   145→      setResult("Erro de conexão com a IA.");
   146→    }
   147→    loadingAction(null);
   148→  }, [topic, thinking, mergeNodes, addEdge]);
   149→
   150→  // Summarize subtree
   151→  const handleSummarize = useCallback(async () => {
   152→    if (!selectedNode) return;
   153→    loadingAction("summarize");
   154→    setResult(null);
   155→    try {
   156→      // Get subtree nodes
   157→      const subtreeIds = new Set<string>();
   158→      const queue = [selectedNode.id];
   159→      while (queue.length) {
   160→        const cur = queue.shift()!;
   161→        if (subtreeIds.has(cur)) continue;
   162→        subtreeIds.add(cur);
   163→        for (const e of edges) {
   164→          if (e.sourceId === cur) queue.push(e.targetId);
   165→          if (e.targetId === cur) queue.push(e.sourceId);
   166→        }
   167→      }
   168→      const subtree = nodes.filter((n) => subtreeIds.has(n.id));
   169→      const res = await fetch("/api/ai/summarize", {
   170→        method: "POST",
   171→        headers: { "Content-Type": "application/json" },
   172→        body: JSON.stringify({
   173→          topic: selectedNode.title,
   174→          nodes: subtree,
   175→          thinking,
   176→        }),
   177→      });
   178→      const data = await res.json();
   179→      setResult(data.summary ?? data.error ?? "Erro ao resumir.");
   180→    } catch (e) {
   181→      setResult("Erro de conexão com a IA.");
   182→    }
   183→    loadingAction(null);
   184→  }, [selectedNode, nodes, edges, thinking]);
   185→
   186→  // Suggest connections
   187→  const handleSuggest = useCallback(async () => {
   188→    loadingAction("suggest");
   189→    setResult(null);
   190→    try {
   191→      const existingPairs: Array<[string, string]> = edges.map((e) => [e.sourceId, e.targetId]);
   192→      const res = await fetch("/api/ai/suggest", {
   193→        method: "POST",
   194→        headers: { "Content-Type": "application/json" },
   195→        body: JSON.stringify({
   196→          nodes: nodes.slice(0, 30).map((n) => ({ id: n.id, title: n.title, kind: n.kind })),
   197→          existingPairs,
   198→          thinking,
   199→        }),
   200→      });
   201→      const data = await res.json();
   202→      if (data.suggestions) {
   203→        let text = "💡 Sugestões de conexões:\n\n";
   204→        for (const s of data.suggestions) {
   205→          const src = nodes.find((n) => n.id === s.source);
   206→          const tgt = nodes.find((n) => n.id === s.target);
   207→          text += `- **${src?.title ?? s.source}** ↔ **${tgt?.title ?? s.target}** (${s.kind})${s.rationale ? ` — ${s.rationale}` : ""}\n`;
   208→        }
   209→        setResult(text);
   210→      } else {
   211→        setResult(data.error ?? "Nenhuma sugestão encontrada.");
   212→      }
   213→    } catch (e) {
   214→      setResult("Erro de conexão com a IA.");
   215→    }
   216→    loadingAction(null);
   217→  }, [nodes, edges, thinking]);
   218→
   219→  // Chat
   220→  const handleChatSend = useCallback(async () => {
   221→    if (!chatInput.trim()) return;
   222→    const userMsg = { role: "user", content: chatInput };
   223→    const newMessages = [...chatMessages, userMsg];
   224→    setChatMessages(newMessages);
   225→    setChatInput("");
   226→    loadingAction("chat");
   227→    try {
   228→      const mapContext = `Mapa: "${mapTitle}". Nós: ${nodes.slice(0, 20).map((n) => `${n.title}(${n.kind})`).join(", ")}. Arestas: ${edges.slice(0, 15).map((e) => `${nodes.find(n=>n.id===e.sourceId)?.title ?? e.sourceId}→${nodes.find(n=>n.id===e.targetId)?.title ?? e.targetId}`).join(", ")}`;
   229→      const res = await fetch("/api/ai/chat", {
   230→        method: "POST",
   231→        headers: { "Content-Type": "application/json" },
   232→        body: JSON.stringify({
   233→          messages: newMessages,
   234→          mapContext,
   235→          thinking,
   236→        }),
   237→      });
   238→      const data = await res.json();
   239→      setChatMessages([...newMessages, { role: "assistant", content: data.reply ?? "Erro." }]);
   240→    } catch (e) {
   241→      setChatMessages([...newMessages, { role: "assistant", content: "Erro de conexão." }]);
   242→    }
   243→    loadingAction(null);
   244→  }, [chatInput, chatMessages, mapTitle, nodes, edges, thinking]);
   245→
   246→  // Auto-scroll chat to bottom
   247→  useEffect(() => {
   248→    if (chatScrollRef.current) {
   249→      chatScrollRef.current.scrollTop = chatScrollRef.current.scrollHeight;
   250→    }
   251→  }, [chatMessages, loading]);
   252→
   253→  // Generate image for selected node
   254→  const handleImage = useCallback(async () => {
   255→    if (!selectedNode) return;
   256→    loadingAction("image");
   257→    setResult(null);
   258→    try {
   259→      const res = await fetch("/api/ai/image", {
   260→        method: "POST",
   261→        headers: { "Content-Type": "application/json" },
   262→        body: JSON.stringify({
   263→          prompt: `Illustration representing "${selectedNode.title}". ${selectedNode.content ?? ""} Clean, modern, minimal style.`,
   264→          size: "1024x1024",
   265→        }),
   266→      });
   267→      const data = await res.json();
   268→      if (data.image) {
   269→        updateNode(selectedNode.id, { image: data.image });
   270→        setResult("✅ Imagem gerada e aplicada ao nó.");
   271→      } else {
   272→        setResult(data.error ?? "Erro ao gerar imagem.");
   273→      }
   274→    } catch (e) {
   275→      setResult("Erro de conexão.");
   276→    }
   277→    loadingAction(null);
   278→  }, [selectedNode, updateNode]);
   279→
   280→  // Auto-layout
   281→  const handleLayout = useCallback(async () => {
   282→    loadingAction("layout");
   283→    setResult(null);
   284→    try {
   285→      const res = await fetch("/api/ai/layout", {
   286→        method: "POST",
   287→        headers: { "Content-Type": "application/json" },
   288→        body: JSON.stringify({
   289→          nodes: nodes.map((n) => ({ id: n.id, parentId: n.parentId })),
   290→          edges: edges.map((e) => ({ sourceId: e.sourceId, targetId: e.targetId })),
   291→          mode: "radial",
   292→        }),
   293→      });
   294→      const data = await res.json();
   295→      if (data.positions) {
   296→        for (const n of nodes) {
   297→          const pos = data.positions[n.id];
   298→          if (pos) {
   299→            updateNode(n.id, { x: pos.x, y: pos.y });
   300→          }
   301→        }
   302→        setResult("✅ Layout reorganizado.");
   303→      } else {
   304→        setResult("Erro no layout.");
   305→      }
   306→    } catch (e) {
   307→      setResult("Erro de conexão.");
   308→    }
   309→    loadingAction(null);
   310→  }, [nodes, edges, updateNode]);
   311→
   312→  if (!open) return null;
   313→
   314→  const tabs: Array<{ id: AITab; icon: React.ReactNode; label: string; enabled: boolean }> = [
   315→    { id: "expand", icon: <Expand className="h-4 w-4" />, label: "Expandir", enabled: settings.ai.expandNode },
   316→    { id: "generate", icon: <Map className="h-4 w-4" />, label: "Gerar mapa", enabled: settings.ai.generateMap },
   317→    { id: "summarize", icon: <FileText className="h-4 w-4" />, label: "Resumir", enabled: settings.ai.summarize },
   318→    { id: "suggest", icon: <Link2 className="h-4 w-4" />, label: "Conexões", enabled: settings.ai.suggestConnections },
   319→    { id: "chat", icon: <MessageSquare className="h-4 w-4" />, label: "Chat", enabled: settings.ai.chatAssistant },
   320→    { id: "image", icon: <ImagePlus className="h-4 w-4" />, label: "Imagem", enabled: settings.ai.generateImage },
   321→  ];
   322→
   323→  // Pick the first enabled tab if current is disabled
   324→  const effectiveTab = tabs.find((t) => t.id === tab && t.enabled) ? tab : tabs.find((t) => t.enabled)?.id ?? "expand";
   325→
   326→  return (
   327→    <div className="w-[340px] bg-card border-l border-border flex flex-col shadow-2xl fade-in z-30">
   328→      {/* header */}
   329→      <div className="flex items-center justify-between px-3 py-2.5 border-b border-border bg-gradient-to-r from-primary/15 via-primary/5 to-transparent">
   330→        <h3 className="text-sm font-semibold flex items-center gap-1.5">
   331→          <div className="h-6 w-6 rounded-md bg-primary/15 flex items-center justify-center">
   332→            <Sparkles className="h-3.5 w-3.5 text-primary" />
   333→          </div>
   334→          Inteligência Artificial
   335→        </h3>
   336→        <Button variant="ghost" size="icon" className="h-7 w-7" onClick={onClose}>
   337→          <X className="h-4 w-4" />
   338→        </Button>
   339→      </div>
   340→
   341→      {/* tab buttons */}
   342→      <div className="flex flex-wrap gap-1 px-2.5 py-2 border-b border-border bg-muted/30">
   343→        {tabs.filter((t) => t.enabled).map((t) => (
   344→          <button
   345→            key={t.id}
   346→            className={`flex items-center gap-1 px-2.5 py-1.5 rounded-md text-xs font-medium transition-all ${
   347→              effectiveTab === t.id
   348→                ? "bg-primary text-primary-foreground shadow-sm"
   349→                : "text-muted-foreground hover:bg-accent hover:text-foreground"
   350→            }`}
   351→            onClick={() => { setTab(t.id); setResult(null); }}
   352→          >
   353→            {t.icon}
   354→            <span className="hidden sm:inline">{t.label}</span>
   355→          </button>
   356→        ))}
   357→      </div>
   358→
   359→      <ScrollArea className="flex-1">
   360→        <div className="p-3 flex flex-col gap-3">
   361→        {/* Expand */}
   362→        {effectiveTab === "expand" && (
   363→          <>
   364→            <p className="text-xs text-muted-foreground leading-relaxed">
   365→              Expande o nó selecionado em conceitos-filho gerados pela IA.
   366→            </p>
   367→            {selectedNode ? (
   368→              <div className="flex items-center gap-2 p-2.5 rounded-lg bg-accent/40 border border-border">
   369→                <div
   370→                  className="h-7 w-7 rounded-md flex items-center justify-center shrink-0"
   371→                  style={{ background: `${NODE_KIND_META[selectedNode.kind]?.color ?? "#10b981"}22`, color: NODE_KIND_META[selectedNode.kind]?.color ?? "#10b981" }}
   372→                >
   373→                  <span className="text-[10px] font-bold">{selectedNode.title[0]}</span>
   374→                </div>
   375→                <span className="text-sm font-medium truncate">{selectedNode.title}</span>
   376→              </div>
   377→            ) : (
   378→              <p className="text-xs text-destructive bg-destructive/10 p-2 rounded-md">⚠ Selecione um nó primeiro.</p>
   379→            )}
   380→            <Button
   381→              size="sm"
   382→              className={`h-8 text-xs gap-1.5 w-full transition-all ${
   383→                !selectedNode || loading !== null ? "opacity-50 grayscale cursor-not-allowed bg-muted text-muted-foreground hover:bg-muted" : ""
   384→              }`}
   385→              disabled={!selectedNode || loading !== null}
   386→              onClick={handleExpand}
   387→            >
   388→              {loading === "expand" ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Expand className="h-3.5 w-3.5" />}
   389→              Expandir nó
   390→            </Button>
   391→          </>
   392→        )}
   393→
   394→        {/* Generate */}
   395→        {effectiveTab === "generate" && (
   396→          <>
   397→            <p className="text-xs text-muted-foreground leading-relaxed">
   398→              Gera um mapa mental completo a partir de um tema.
   399→            </p>
   400→            <Input
   401→              value={topic}
   402→              onChange={(e) => setTopic(e.target.value)}
   403→              onKeyDown={(e) => { if (e.key === "Enter" && topic.trim() && !loading) handleGenerate(); }}
   404→              placeholder="Ex: Arquitetura de microsserviços..."
   405→              className="h-9 text-sm"
   406→            />
   407→            <div className="flex flex-wrap gap-1">
   408→              {["Inteligência Artificial", "Mudanças Climáticas", "Filosofia Grega", "Nutrição Humana"].map((s) => (
   409→                <button
   410→                  key={s}
   411→                  className="text-[10px] px-2 py-0.5 rounded-full border border-border bg-muted/40 hover:bg-accent hover:border-primary/40 text-muted-foreground transition-colors"
   412→                  onClick={() => setTopic(s)}
   413→                >
   414→                  {s}
   415→                </button>
   416→              ))}
   417→            </div>
   418→            <Button
   419→              size="sm"
   420→              className={`h-8 text-xs gap-1.5 w-full transition-all ${
   421→                !topic.trim() || loading !== null ? "opacity-50 grayscale cursor-not-allowed bg-muted text-muted-foreground hover:bg-muted" : ""
   422→              }`}
   423→              disabled={!topic.trim() || loading !== null}
   424→              onClick={handleGenerate}
   425→            >
   426→              {loading === "generate" ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Map className="h-3.5 w-3.5" />}
   427→              Gerar mapa
   428→            </Button>
   429→          </>
   430→        )}
   431→
   432→        {/* Summarize */}
   433→        {effectiveTab === "summarize" && (
   434→          <>
   435→            <p className="text-xs text-muted-foreground leading-relaxed">
   436→              Resume o ramo do nó selecionado em texto Markdown.
   437→            </p>
   438→            {selectedNode ? (
   439→              <div className="flex items-center gap-2 p-2.5 rounded-lg bg-accent/40 border border-border">
   440→                <FileText className="h-4 w-4 text-primary shrink-0" />
   441→                <span className="text-sm font-medium truncate">{selectedNode.title}</span>
   442→              </div>
   443→            ) : (
   444→              <p className="text-xs text-destructive bg-destructive/10 p-2 rounded-md">⚠ Selecione um nó primeiro.</p>
   445→            )}
   446→            <Button
   447→              size="sm"
   448→              className={`h-8 text-xs gap-1.5 w-full transition-all ${
   449→                !selectedNode || loading !== null ? "opacity-50 grayscale cursor-not-allowed bg-muted text-muted-foreground hover:bg-muted" : ""
   450→              }`}
   451→              disabled={!selectedNode || loading !== null}
   452→              onClick={handleSummarize}
   453→            >
   454→              {loading === "summarize" ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <FileText className="h-3.5 w-3.5" />}
   455→              Resumir subárvore
   456→            </Button>
   457→          </>
   458→        )}
   459→
   460→        {/* Suggest */}
   461→        {effectiveTab === "suggest" && (
   462→          <>
   463→            <p className="text-xs text-muted-foreground leading-relaxed">
   464→              Sugere conexões entre nós que ainda não estão ligados.
   465→            </p>
   466→            <div className="rounded-lg bg-muted/30 border border-border p-2.5 text-xs text-muted-foreground">
   467→              <p className="flex items-center gap-1.5">
   468→                <Link2 className="h-3.5 w-3.5" />
   469→                <strong className="text-foreground">{nodes.length}</strong> nós · <strong className="text-foreground">{edges.length}</strong> conexões
   470→              </p>
   471→            </div>
   472→            <Button
   473→              size="sm"
   474→              className={`h-8 text-xs gap-1.5 w-full transition-all ${
   475→                nodes.length < 2 || loading !== null ? "opacity-50 grayscale cursor-not-allowed bg-muted text-muted-foreground hover:bg-muted" : ""
   476→              }`}
   477→              disabled={nodes.length < 2 || loading !== null}
   478→              onClick={handleSuggest}
   479→            >
   480→              {loading === "suggest" ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Link2 className="h-3.5 w-3.5" />}
   481→              Sugerir conexões
   482→            </Button>
   483→          </>
   484→        )}
   485→
   486→        {/* Chat */}
   487→        {effectiveTab === "chat" && (
   488→          <>
   489→            <div className="flex items-center justify-between">
   490→              <p className="text-xs text-muted-foreground">Converse com a IA sobre o mapa.</p>
   491→              {chatMessages.length > 0 && (
   492→                <button
   493→                  className="text-[10px] text-muted-foreground hover:text-destructive flex items-center gap-1 transition-colors"
   494→                  onClick={() => setChatMessages([])}
   495→                  title="Limpar conversa"
   496→                >

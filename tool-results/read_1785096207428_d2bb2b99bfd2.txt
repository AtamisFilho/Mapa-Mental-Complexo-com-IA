   394→  organizeLayout: () => {
   395→    const { nodes, edges } = get();
   396→    if (nodes.length === 0) return;
   397→    // Simple collision avoidance + radial layout
   398→    // 1. Find roots (nodes with no incoming edges)
   399→    const incoming = new Set(edges.map((e) => e.targetId));
   400→    const roots = nodes.filter((n) => !incoming.has(n.id));
   401→    const rootId = roots.length > 0 ? roots[0].id : nodes[0].id;
   402→    // 2. Build parent-child map
   403→    const childrenOf = new Map<string, string[]>();
   404→    for (const e of edges) {
   405→      if (!childrenOf.has(e.sourceId)) childrenOf.set(e.sourceId, []);
   406→      childrenOf.get(e.sourceId)!.push(e.targetId);
   407→    }
   408→    // 3. BFS to assign depth levels
   409→    const depthMap = new Map<string, number>();
   410→    const queue: Array<{ id: string; depth: number }> = [{ id: rootId, depth: 0 }];
   411→    const visited = new Set<string>();
   412→    while (queue.length) {
   413→      const { id, depth } = queue.shift()!;
   414→      if (visited.has(id)) continue;
   415→      visited.add(id);
   416→      depthMap.set(id, depth);
   417→      for (const child of childrenOf.get(id) ?? []) {
   418→        if (!visited.has(child)) queue.push({ id: child, depth: depth + 1 });
   419→      }
   420→    }
   421→    // Assign depth 0 to unvisited nodes
   422→    for (const n of nodes) {
   423→      if (!depthMap.has(n.id)) depthMap.set(n.id, 0);
   424→    }
   425→    // 4. Compute positions by depth level with radial spread
   426→    const maxDepth = Math.max(...Array.from(depthMap.values()));
   427→    const ringRadii = [0, 260, 460, 620, 760, 900, 1040];
   428→    const centerX = 400;
   429→    const centerY = 300;
   430→    // Group nodes by depth
   431→    const byDepth = new Map<number, string[]>();
   432→    for (const [id, d] of depthMap) {
   433→      if (!byDepth.has(d)) byDepth.set(d, []);
   434→      byDepth.get(d)!.push(id);
   435→    }
   436→    const newPositions = new Map<string, { x: number; y: number }>();
   437→    for (const [d, ids] of byDepth) {
   438→      const radius = d === 0 ? 0 : (ringRadii[d] ?? ringRadii[ringRadii.length - 1] + (d - ringRadii.length + 1) * 140);
   439→      const angleStep = (2 * Math.PI) / Math.max(ids.length, 1);
   440→      const startAngle = d === 0 ? 0 : -Math.PI / 2;
   441→      for (let i = 0; i < ids.length; i++) {
   442→        const angle = startAngle + i * angleStep;
   443→        const node = nodes.find((n) => n.id === ids[i]);
   444→        const w = node?.width ?? 220;
   445→        const h = node?.height ?? 88;
   446→        newPositions.set(ids[i], {
   447→          x: centerX + Math.cos(angle) * radius - w / 2,
   448→          y: centerY + Math.sin(angle) * radius - h / 2,
   449→        });
   450→      }
   451→    }
   452→    // 5. Simple collision resolution: push overlapping nodes apart
   453→    const iterations = 3;
   454→    for (let iter = 0; iter < iterations; iter++) {
   455→      for (let i = 0; i < nodes.length; i++) {
   456→        const posA = newPositions.get(nodes[i].id) ?? { x: nodes[i].x, y: nodes[i].y };
   457→        for (let j = i + 1; j < nodes.length; j++) {
   458→          const posB = newPositions.get(nodes[j].id) ?? { x: nodes[j].x, y: nodes[j].y };
   459→          const wA = nodes[i].width + 20; // padding
   460→          const hA = nodes[i].height + 20;
   461→          const wB = nodes[j].width + 20;
   462→          const hB = nodes[j].height + 20;
   463→          const overlapX = (wA / 2 + wB / 2) - Math.abs(posA.x + nodes[i].width / 2 - posB.x + nodes[j].width / 2);
   464→          const overlapY = (hA / 2 + hB / 2) - Math.abs(posA.y + nodes[i].height / 2 - posB.y + nodes[j].height / 2);
   465→          if (overlapX > 0 && overlapY > 0) {
   466→            // Push apart along the axis with less overlap
   467→            const push = overlapX < overlapY ? overlapX / 2 : overlapY / 2;
   468→            const dirX = posA.x + nodes[i].width / 2 < posB.x + nodes[j].width / 2 ? -1 : 1;
   469→            const dirY = posA.y + nodes[i].height / 2 < posB.y + nodes[j].height / 2 ? -1 : 1;
   470→            if (overlapX < overlapY) {
   471→              newPositions.set(nodes[i].id, { ...posA, x: posA.x + dirX * push });
   472→              newPositions.set(nodes[j].id, { ...posB, x: posB.x - dirX * push });
   473→            } else {
   474→              newPositions.set(nodes[i].id, { ...posA, y: posA.y + dirY * push });
   475→              newPositions.set(nodes[j].id, { ...posB, y: posB.y - dirY * push });
   476→            }
   477→          }
   478→        }
   479→      }
   480→    }
   481→    // 6. Apply new positions
   482→    const updated = nodes.map((n) => {
   483→      const pos = newPositions.get(n.id);
   484→      return pos ? { ...n, x: pos.x, y: pos.y, updatedAt: new Date().toISOString() } : n;
   485→    });
   486→    set({ nodes: updated, dirty: true });
   487→  },

   489→  insertSubtree: (template, position, parentId) => {
   490→    // Push current state to history so the insertion is undoable as a single step
   491→    get().pushHistory();
   492→
   493→    // Tree layout: root at `position`. Children are indented horizontally by
   494→    // 200px and stacked vertically with 60px spacing. Each child's subtree is
   495→    // placed below the previous sibling's subtree (so siblings don't overlap).
   496→    const NODE_HEIGHT = 80;
   497→    const NODE_WIDTH = 200;
   498→    const VERTICAL_GAP = 60;
   499→    const HORIZONTAL_INDENT = 200;
   500→
   501→    const addNode = get().addNode;
   502→    const addEdge = get().addEdge;
   503→
   504→    // Recursive helper — returns the y-coordinate of the BOTTOM of the inserted
   505→    // subtree (so the next sibling can be placed below without overlap).
   506→    const buildSubtree = (
   507→      node: SubtreeTemplateNode,
   508→      x: number,
   509→      y: number,
   510→      parent: string | null
   511→    ): number => {
   512→      const id = addNode({
   513→        title: node.title,
   514→        kind: node.kind,
   515→        content: node.content,
   516→        note: node.note,
   517→        icon: node.icon,
   518→        x,
   519→        y,
   520→        width: NODE_WIDTH,
   521→        height: NODE_HEIGHT,
   522→        parentId: parent,
   523→      });
   524→      if (parent) {
   525→        addEdge(parent, id);
   526→      }
   527→
   528→      let bottomY = y + NODE_HEIGHT;
   529→      const kids = node.children;
   530→      if (kids && kids.length > 0) {
   531→        let childY = y + NODE_HEIGHT + VERTICAL_GAP;
   532→        for (const child of kids) {
   533→          const childBottom = buildSubtree(
   534→            child,
   535→            x + HORIZONTAL_INDENT,
   536→            childY,
   537→            id
   538→          );
   539→          childY = childBottom + VERTICAL_GAP;
   540→          if (childBottom > bottomY) bottomY = childBottom;
   541→        }
   542→      }
   543→      return bottomY;
   544→    };
   545→
   546→    const rootId = addNode({
   547→      title: template.title,
   548→      kind: template.kind,
   549→      content: template.content,
   550→      note: template.note,
   551→      icon: template.icon,
   552→      x: position.x,
   553→      y: position.y,
   554→      width: NODE_WIDTH,
   555→      height: NODE_HEIGHT,
   556→      parentId: parentId ?? null,
   557→    });
   558→    if (parentId) {
   559→      addEdge(parentId, rootId);
   560→    }
   561→
   562→    // Lay out the root's children beneath it
   563→    const kids = template.children;
   564→    if (kids && kids.length > 0) {
   565→      let childY = position.y + NODE_HEIGHT + VERTICAL_GAP;
   566→      for (const child of kids) {
   567→        const childBottom = buildSubtree(
   568→          child,
   569→          position.x + HORIZONTAL_INDENT,
   570→          childY,
   571→          rootId
   572→        );
   573→        childY = childBottom + VERTICAL_GAP;
   574→      }
   575→    }
   576→
   577→    // Select the new root so the user can immediately interact with it
   578→    set({ selectedNodeIds: [rootId], selectedEdgeIds: [], dirty: true });
   579→    return rootId;
   580→  },
   581→
   582→  pushHistory: () =>
   583→    set((s) => {
   584→      const entry: HistoryEntry = { nodes: s.nodes, edges: s.edges };
   585→      const past = [...s.past, entry].slice(-50);
   586→      return { past, future: [] };
   587→    }),
   588→
   589→  undo: () =>
   590→    set((s) => {
   591→      if (s.past.length === 0) return {};
   592→      const previous = s.past[s.past.length - 1];
   593→      const current: HistoryEntry = { nodes: s.nodes, edges: s.edges };
   594→      return {
   595→        past: s.past.slice(0, -1),
   596→        future: [current, ...s.future].slice(0, 50),
   597→        nodes: previous.nodes,
   598→        edges: previous.edges,
   599→        dirty: true,
   600→      };
   601→    }),
   602→
   603→  redo: () =>
   604→    set((s) => {
   605→      if (s.future.length === 0) return {};
   606→      const next = s.future[0];
   607→      const current: HistoryEntry = { nodes: s.nodes, edges: s.edges };
   608→      return {
   609→        past: [...s.past, current].slice(-50),
   610→        future: s.future.slice(1),
   611→        nodes: next.nodes,
   612→        edges: next.edges,
   613→        dirty: true,
   614→      };
   615→    }),
   616→
   617→  // ── Reparent (Task 16-B) ──────────────────────────────────────────────
   618→  setReparentTarget: (id) => set({ reparentTargetId: id }),
   619→  setDraggedNode: (id) => set({ draggedNodeId: id }),
   620→
   621→  isDescendantOf: (nodeId, potentialAncestorId) => {
   622→    // Check if `nodeId` is a descendant of `potentialAncestorId` by traversing
   623→    // edges downward. Returns true if the node is in the subtree of the ancestor,
   624→    // which would create a cycle if we reparented the ancestor under the node.
   625→    const { edges } = get();
   626→    // Build children map (source → targets)
   627→    const childrenOf = new Map<string, string[]>();
   628→    for (const e of edges) {
   629→      if (!childrenOf.has(e.sourceId)) childrenOf.set(e.sourceId, []);
   630→      childrenOf.get(e.sourceId)!.push(e.targetId);
   631→    }
   632→    // BFS from potentialAncestorId
   633→    const visited = new Set<string>();
   634→    const queue = [potentialAncestorId];
   635→    while (queue.length) {
   636→      const cur = queue.shift()!;
   637→      if (cur === nodeId) return true; // nodeId is a descendant
   638→      if (visited.has(cur)) continue;
   639→      visited.add(cur);
   640→      for (const child of childrenOf.get(cur) ?? []) {
   641→        if (!visited.has(child)) queue.push(child);
   642→      }
   643→    }
   644→    return false;
   645→  },
   646→
   647→  reparentNode: (nodeId, newParentId) => {
   648→    // Cannot reparent to self
   649→    if (nodeId === newParentId) return false;
   650→    // Cannot reparent to own descendant (would create cycle)
   651→    if (get().isDescendantOf(newParentId, nodeId)) return false;
   652→
   653→    // Note: pushHistory() is NOT called here because the caller
   654→    // (handleNodePointerDown in MindMapCanvas) already pushes history
   655→    // before the drag starts, which captures the initial state.
   656→    // The reparent + position changes should be undoable as a single step.
   657→
   658→    // Find the old parent edge (where targetId === nodeId)
   659→    const oldEdge = get().edges.find((e) => e.targetId === nodeId);
   660→    const oldEdgeKind = oldEdge?.kind ?? "related";
   661→
   662→    // Delete old parent edge if exists
   663→    if (oldEdge) {
   664→      get().deleteEdge(oldEdge.id);
   665→    }
   666→
   667→    // Create new edge from newParentId to nodeId
   668→    get().addEdge(newParentId, nodeId, oldEdgeKind as EdgeKind);
   669→
   670→    // Optionally reposition node near its new parent (+200px right, +60px below)
   671→    const newParent = get().nodes.find((n) => n.id === newParentId);
   672→    if (newParent) {
   673→      const draggedNode = get().nodes.find((n) => n.id === nodeId);
   674→      if (draggedNode) {
   675→        // Check if there are already children of the new parent at that position
   676→        // and offset further to avoid overlap
   677→        const childrenOfNewParent = get().edges
   678→          .filter((e) => e.sourceId === newParentId && e.targetId !== nodeId)
   679→          .map((e) => get().nodes.find((n) => n.id === e.targetId));
   680→        // Calculate y offset — stack below existing children
   681→        let yOffset = 60;
   682→        for (const child of childrenOfNewParent) {
   683→          if (child) {
   684→            yOffset = Math.max(yOffset, child.y - newParent.y + child.height + 60);
   685→          }
   686→        }
   687→        get().updateNode(nodeId, {
   688→          x: newParent.x + newParent.width + 200,
   689→          y: newParent.y + yOffset,
   690→        });
   691→      }
   692→    }
   693→
   694→    // Clear reparent state
   695→    set({ reparentTargetId: null, draggedNodeId: null });
   696→    return true;
   697→  },
   698→
   699→  // ── Search & Replace (Task 15-B) ─────────────────────────────────────
   700→  setSearchQuery: (q) => set({ searchQuery: q }),
   701→  setSearchMatches: (ids) => set({ searchMatches: ids }),
   702→  setHighlightedMatch: (id) => set({ highlightedMatchId: id }),
   703→
   704→  searchNodes: (query, opts) => {
   705→    const caseSensitive = opts?.caseSensitive ?? false;
   706→    const titleOnly = opts?.titleOnly ?? false;
   707→    const q = caseSensitive ? query : query.toLowerCase();
   708→    if (!query.trim()) {
   709→      set({ searchQuery: query, searchMatches: [], highlightedMatchId: null });
   710→      return [];
   711→    }
   712→    const nodes = get().nodes;
   713→    const matched = nodes
   714→      .filter((n) => {
   715→        const title = caseSensitive ? n.title : n.title.toLowerCase();
   716→        if (title.includes(q)) return true;
   717→        if (titleOnly) return false;
   718→        const content = caseSensitive ? (n.content ?? "") : (n.content ?? "").toLowerCase();
   719→        const note = caseSensitive ? (n.note ?? "") : (n.note ?? "").toLowerCase();
   720→        return content.includes(q) || note.includes(q);
   721→      })
   722→      .map((n) => n.id);
   723→    set({
   724→      searchQuery: query,
   725→      searchMatches: matched,
   726→      highlightedMatchId: matched.length > 0 ? matched[0] : null,
   727→    });
   728→    return matched;
   729→  },
   730→
   731→  replaceInNode: (nodeId, search, replacement, opts) => {
   732→    const caseSensitive = opts?.caseSensitive ?? false;
   733→    if (!search) return 0;
   734→    const node = get().nodes.find((n) => n.id === nodeId);
   735→    if (!node) return 0;
   736→    // Build a case-sensitive or case-insensitive regex; escape regex special chars.
   737→    const flags = caseSensitive ? "g" : "gi";
   738→    const escaped = search.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   739→    const re = new RegExp(escaped, flags);
   740→    let count = 0;
   741→    const patch: Partial<MapNode> = {};
   742→    if (node.title) {
   743→      const matches = node.title.match(re);
   744→      if (matches && matches.length > 0) {
   745→        patch.title = node.title.replace(re, replacement);
   746→        count += matches.length;
   747→      }
   748→    }
   749→    if (node.content) {
   750→      const matches = node.content.match(re);
   751→      if (matches && matches.length > 0) {
   752→        patch.content = node.content.replace(re, replacement);
   753→        count += matches.length;
   754→      }
   755→    }
   756→    if (count > 0) {
   757→      get().pushHistory();
   758→      get().updateNode(nodeId, patch);
   759→    }
   760→    return count;
   761→  },
   762→
   763→  replaceAll: (search, replacement, opts) => {
   764→    const caseSensitive = opts?.caseSensitive ?? false;
   765→    if (!search) return 0;
   766→    const matchedIds = get().searchMatches;
   767→    if (matchedIds.length === 0) return 0;
   768→    const flags = caseSensitive ? "g" : "gi";
   769→    const escaped = search.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
   770→    const re = new RegExp(escaped, flags);
   771→    let total = 0;
   772→    const nodes = get().nodes;
   773→    const updates = new Map<string, Partial<MapNode>>();
   774→    for (const id of matchedIds) {
   775→      const n = nodes.find((x) => x.id === id);
   776→      if (!n) continue;
   777→      const patch: Partial<MapNode> = {};
   778→      if (n.title) {
   779→        const m = n.title.match(re);
   780→        if (m && m.length > 0) {
   781→          patch.title = n.title.replace(re, replacement);
   782→          total += m.length;
   783→        }
   784→      }
   785→      if (n.content) {
   786→        const m = n.content.match(re);
   787→        if (m && m.length > 0) {
   788→          patch.content = n.content.replace(re, replacement);
   789→          total += m.length;
   790→        }
   791→      }
   792→      if (Object.keys(patch).length > 0) {
   793→        updates.set(id, patch);
   794→      }
   795→    }
   796→    if (updates.size > 0) {
   797→      get().pushHistory();
   798→      set((s) => ({
   799→        nodes: s.nodes.map((n) =>
   800→          updates.has(n.id)
   801→            ? { ...n, ...updates.get(n.id)!, updatedAt: new Date().toISOString() }
   802→            : n
   803→        ),
   804→        dirty: true,
   805→      }));
   806→    }
   807→    return total;
   808→  },
   809→}));
